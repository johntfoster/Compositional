\section{Material form of conservation of mass}
Given a differential mass ${\rm d}m^\al_\xi$ of component $\al$ in phase $\xi$, define the component partial density per unit current mixture volume by
%
\[
  \rho^\al_\xi(\mbf{x}, t) = \frac{{\rm d}m^\al_\xi}{\dv},
\]
%
where $\dv = {\rm d}x_1 {\rm d}x_2 {\rm d}x_3$ is a current-configuration volume element of the mixture.  The current configuration is a common mixture volume; it is not decomposed into separate current phase bodies.  Each phase $\xi$ is instead assigned its own reference coordinates $\mbf{X}_\xi$ and reference volume element
%
\[
  \dVxi = ({\rm d}X_1)_\xi({\rm d}X_2)_\xi({\rm d}X_3)_\xi.
\]
%
The phase reference coordinates are related to the common current coordinates by a one-to-one, invertible, and continuously differentiable map on the region where that phase is present,
%
\[
  \mbf{x} = \bs{\chi}_\xi(\mbf{X}_\xi,t),
\]
%
with phase velocity $\mbf{v}_\xi = \partial \bs{\chi}_\xi/\partial t$ and Jacobian
%
\[
  J_\xi = \det\left(\p{\bs{\chi}_\xi}{\mbf{X}_\xi}\right).
\]
%
Equivalently, the phase motion is understood on the support
\[
  \Omega_\xi(t) = \{\mbf{x}: \phi_\xi(\mbf{x},t)>0\}.
\]
When a phase locally disappears, the corresponding balance law is interpreted in its weak extended form with $\phi_\xi=0$ and nonnegative phase volume fraction; phase equilibrium and constitutive relations involving that phase are then imposed only where the phase is present.
%
Let $\rho_{\xi0}^\al(\mbf{X}_\xi)$ denote the initial component partial density pulled back to the phase reference coordinates, and let $c^\al_\xi(\mbf{X}_\xi,t)$ denote the cumulative mass of component $\al$ added to phase $\xi$ per unit phase reference volume by reactions and/or phase transformations.  Conservation of mass for component $\al$ in phase $\xi$ over an arbitrary material region can then be written as
%
\begin{align}
  \int_{\mc{B}_{\xi 0}} \left(\rho_{\xi 0}^\al + c^\al_\xi\right)\dVxi
  &= \int_{\mc{B}} \rho_\xi^\al {\rm d}v \notag \\
  &= \int_{\mc{B}_{\xi 0}} \rho_{\xi}^\al J_\xi \dVxi .
  \label{eq:material_mass}
\end{align}
%
Here $\mc{B}$ is the common current mixture region corresponding, through the phase-$\xi$ map, to the phase reference region $\mc{B}_{\xi0}$.  Since \eqref{eq:material_mass} must hold for arbitrary $\mc{B}_{\xi0}$,
%
\begin{equation}
  \rho_{\xi 0}^\alpha + c^\alpha_\xi = \rho_\xi^\alpha J_\xi.
  \label{eq:material_mass_2}
\end{equation}
%
Taking the material time derivative following phase $\xi$ gives
%
\begin{equation}
  \frac{D_\xi}{Dt}\left(\rho_\xi^\alpha J_\xi\right) = \dot{c}^\alpha_\xi,
\end{equation}
%
where $\dot{c}^\alpha_\xi = D_\xi c^\alpha_\xi/Dt$.  Using
%
\[
  \frac{D_\xi J_\xi}{Dt} = J_\xi \nabla_\mbf{x}\cdot\mbf{v}_\xi,
\]
%
we obtain
%
\begin{align}
  \frac{D_\xi \rho_\xi^\alpha}{Dt}
  + \rho_\xi^\alpha \nabla_\mbf{x}\cdot\mbf{v}_\xi
  &= \frac{\dot{c}^\alpha_\xi}{J_\xi} .
  \label{eq:component_material_mass}
\end{align}
%
We define the current production rate per unit current mixture volume by
%
\[
  \hat{c}^\al_\xi(\mbf{x},t) = \frac{\dot{c}^\al_\xi}{J_\xi}.
\]
%
The material time derivative of a quantity $\square_\xi$ following the velocity of phase $\zeta$ is
%
\[
  \frac{D_\zeta \square_\xi}{Dt}
  = \frac{\partial \square_\xi}{\partial t}
  + \mbf{v}_\zeta\cdot\nabla_{\mbf{x}}\square_\xi,
\]
%
and we use the shorthand $\dot\square_\xi = D_\xi \square_\xi/Dt$ for the derivative following phase $\xi$.  Expanding \eqref{eq:component_material_mass} gives the spatial component mass balance
%
\begin{align}
  \frac{\partial \rho_\xi^\alpha}{\partial t}
  + \nabla_\mbf{x}\cdot\left(\rho_\xi^\alpha\mbf{v}_\xi\right)
  = \hat{c}^\alpha_\xi .
  \label{eq:component_spatial_mass}
\end{align}
%
Now introduce the phase partial density
%
\[
  \rho_\xi = \sum_\alpha \rho_\xi^\alpha,
\]
%
and the mass fraction of component $\alpha$ in phase $\xi$,
%
\begin{equation}
  \eta^\alpha_\xi = \frac{\rho^\alpha_\xi}{\rho_\xi}.
  \label{eq:phase_mass_fraction}
\end{equation}
%
Thus
%
\[
  \rho_\xi^\alpha = \rho_\xi \eta^\alpha_\xi,
  \qquad
  \sum_\alpha \eta^\alpha_\xi = 1.
\]
%
The phase volume fraction is defined by
%
\[
  \phi_\xi = \frac{\dv_\xi}{\dv},
\]
%
where $\dv_\xi$ is a bookkeeping measure of the partial volume occupied by phase $\xi$ in the current mixture volume $\dv$.  The intrinsic density is introduced only at the phase level,
%
\[
  \bar\rho_\xi = \frac{{\rm d}m_\xi}{\dv_\xi},
\]
%
so that
%
\[
  \rho_\xi = \phi_\xi \bar\rho_\xi,
  \qquad
  \rho_\xi^\alpha = \phi_\xi \bar\rho_\xi \eta^\alpha_\xi.
\]
%
Substituting this representation into \eqref{eq:component_spatial_mass} gives
%
\begin{align}
  \frac{\partial}{\partial t}\left(\phi_\xi \bar\rho_\xi \eta^\al_\xi\right)
  + \nabla_\mbf{x}\cdot\left(\phi_\xi \bar\rho_\xi \eta^\al_\xi \mbf{v}_\xi\right)
  = \hat{c}^\alpha_\xi .
  \label{eq:spatial_mass_2}
\end{align}
%
Summing over $\alpha$ and using $\sum_\alpha \eta_\xi^\alpha=1$ gives the phase mass balance
%
\begin{align}
  \frac{\partial}{\partial t}\left(\phi_\xi \bar\rho_\xi\right)
  + \nabla_\mbf{x}\cdot\left(\phi_\xi \bar\rho_\xi \mbf{v}_\xi\right)
  = \sum_\alpha \hat{c}^\al_\xi .
  \label{eq:phase_spatial_mass}
\end{align}
%
Finally, summing over all phases gives the overall mixture mass balance
%
\begin{align}
  \frac{\partial \rho}{\partial t}
  + \nabla_\mbf{x}\cdot\left(\rho\mbf{v}\right)
  = \sum_\xi \sum_\alpha \hat{c}^\alpha_\xi,
  \label{eq:overall_mass}
\end{align}
%
where
%
\[
  \rho = \sum_\xi \rho_\xi = \sum_\xi \phi_\xi \bar\rho_\xi,
  \qquad
  \rho\mbf{v} = \sum_\xi \rho_\xi\mbf{v}_\xi
  = \sum_\xi \phi_\xi\bar\rho_\xi\mbf{v}_\xi.
\]
%
Equivalently, the barycentric mixture velocity is
%
\[
  \mbf{v} = \frac{1}{\rho}\sum_\xi \rho_\xi\mbf{v}_\xi.
\]
%
For reactions and phase transformations internal to the mixture, total mass is neither created nor destroyed.  Therefore the source terms must satisfy
%
\begin{align}
  \sum_\xi \sum_\alpha \hat{c}^\alpha_\xi = 0,
  \label{eq:overall_mass_constraint}
\end{align}
%
and \eqref{eq:overall_mass} reduces to the conservative mixture mass balance
%
\[
  \frac{\partial \rho}{\partial t}
  + \nabla_\mbf{x}\cdot\left(\rho\mbf{v}\right) = 0.
\]
