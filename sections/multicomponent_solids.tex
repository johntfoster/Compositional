% !TeX root = ../main.tex
\section{Constitutive models and the entropy inequality}
\label{sec:multicomponent_solid_skeletons}
\label{sec:multicomponent_solid_mixture_rules}
\label{sec:constitutive_models_entropy_inequality}

The phase fields introduced in \sref{virtual_power_derivation} allow every
phase to carry component mass fractions.  The solid phases retain that full compositional
structure.  Each \(s\in\mc S\) is a load-bearing multicomponent solid phase with
mass fractions \(\eta_s^\alpha\), while all solid phases share the common
skeleton trajectory
\[
	\mbf x=\bs\chi(\mbf X,t),
	\qquad
	\mbf F=\nabla_{\mbf X}\bs\chi,
	\qquad
	J=\det\mbf F .
\]
The common displacement means that adsorbed species, host minerals, hydrate
cages, bound water, and hydrate guest molecules do not receive separate solid
velocities.  Their mechanical effect enters through solid composition,
stress-free swelling, thermal expansion, elastic moduli, and plastic flow
properties.

\subsection{Solid kinematics and state variables}
\label{sec:MC_solid_kinematics}

Extending Drumheller's distension/true-deformation split for reacting mixtures
\cite{drumheller2000} and its porous-media form in Foster et
al.~\cite{foster2026reactingporous}, the solid kinematic split is
phase-specific but uses the common skeleton deformation,
%
\begin{equation}
	\mbf F=\mbf A_s\bar{\mbf F}_s,
	\qquad
	J=a_s\bar J_s,
	\qquad
	a_s=\det\mbf A_s,
	\qquad
	\bar J_s=\det\bar{\mbf F}_s,
	\qquad
	s\in\mc S .
	\label{eq:MC_solid_distension_decomposition}
\end{equation}
%
The distension tensor \(\mbf A_s\) stores pore-space allocation, packing,
stress-free swelling, and reaction-induced accommodation between the common
skeleton motion and the true solid deformation \(\bar{\mbf F}_s\).  Following
Drumheller's elastic--plastic treatment of true deformation and distension
\cite{drumheller2000}, the elastic, plastic, and stress-free factors are
%
\begin{subequations}
	\label{eq:MC_solid_kinematic_splits}
	\begin{align}
		\mbf A_s
		 & =
		\mbf A_s^e\mbf A_s^p\mbf A_s^0,
		\label{eq:MC_solid_distension_split} \\
		\bar{\mbf F}_s
		 & =
		\bar{\mbf F}_s^e
		\bar{\mbf F}_s^p
		\bar{\mbf F}_s^0 .
		\label{eq:MC_solid_true_split}
	\end{align}
\end{subequations}
%
Here \(\mbf A_s^e\) and \(\bar{\mbf F}_s^e\) are elastic factors,
\(\mbf A_s^p\) and \(\bar{\mbf F}_s^p\) are plastic internal variables, and
\(\mbf A_s^0\) and \(\bar{\mbf F}_s^0\) are reversible stress-free maps that can
include temperature and solid-composition effects,
%
\begin{equation}
	\mbf A_s^0
	=
	\mbf A_s^0(\theta,\eta_s^\beta),
	\qquad
	\bar{\mbf F}_s^0
	=
	\bar{\mbf F}_s^0(\theta,\eta_s^\beta).
	\label{eq:MC_solid_stress_free_maps}
\end{equation}
%
The stress-free maps represent adsorption swelling, desorption shrinkage,
hydrate-cage volume changes, and composition-dependent thermal expansion.
Thermo-poroelastic models make the same physical distinction by assigning pore
volume, fluid, and grain thermal expansions separately
\cite{palciauskas1982thermally,ghabezloo2009stress,ghabezloo2011micromechanics}.
As in standard finite-strain multiplicative plasticity
\cite{simo1988framework,simo1998} and Drumheller's mixture treatment
\cite{drumheller2000}, the true-plastic flow is assumed isochoric,
%
\begin{equation}
	\det\bar{\mbf F}_s^p=1,
	\qquad
	\operatorname{tr}\left[
		\dot{\bar{\mbf F}}_s^p
		(\bar{\mbf F}_s^p)^{-1}
		\right]=0,
	\qquad
	s\in\mc S .
	\label{eq:MC_solid_isochoric_plastic_flow}
\end{equation}
%
The dependent elastic arguments are
%
\begin{equation}
	\mbf A_s^e
	=
	\mbf A_s
	(\mbf A_s^0)^{-1}
	(\mbf A_s^p)^{-1},
	\qquad
	\bar{\mbf F}_s^e
	=
	\bar{\mbf F}_s
	(\bar{\mbf F}_s^0)^{-1}
	(\bar{\mbf F}_s^p)^{-1}.
	\label{eq:MC_solid_elastic_arguments}
\end{equation}
%

The solid component partial densities and mass fractions are
%
\begin{equation}
	\rho_s^\alpha
	=
	\rho_s\eta_s^\alpha
	=
	\phi_s\bar\rho_s\eta_s^\alpha,
	\qquad
	\sum_\alpha\eta_s^\alpha=1,
	\qquad
	\sum_\alpha\dot\eta_s^\alpha=0 .
	\label{eq:MC_solid_component_partial_density}
\end{equation}
%
When component intrinsic densities \(\bar\rho_s^\alpha\) and component solid
volume fractions \(\phi_s^\alpha\) are useful, they satisfy
%
\begin{subequations}
	\label{eq:MC_solid_density_mixture_rules}
	\begin{align}
		\bar\rho_s
		 & =
		\sum_\alpha
		\phi_s^\alpha\bar\rho_s^\alpha,
		\label{eq:MC_solid_intrinsic_density_rule} \\
		\eta_s^\alpha
		 & =
		\frac{\phi_s^\alpha\bar\rho_s^\alpha}{\bar\rho_s},
		\label{eq:MC_solid_mass_fraction_rule}
	\end{align}
	\begin{equation}
		\phi_s^\alpha
		=
		\frac{\eta_s^\alpha\bar\rho_s}{\bar\rho_s^\alpha},
		\qquad
		\sum_\alpha\phi_s^\alpha=1 .
		\label{eq:MC_solid_component_volume_fraction_rule}
	\end{equation}
\end{subequations}
%

The component material balances include both conversion and relative-transport
accumulations,
%
\begin{equation}
	J\phi_s\bar\rho_s\eta_s^\alpha
	=
	\phi_{s0}\bar\rho_{s0}\eta_{s0}^\alpha
	+\mc C_s^\alpha
	+\mc D_s^\alpha,
	\qquad
	s\in\mc S .
	\label{eq:MC_solid_component_material_mass}
\end{equation}
%
Since \(\sum_\alpha\mc D_s^\alpha=0\) by
\eqref{eq:transport_accumulation_zero_sum}, summing over components gives
%
\begin{equation}
	J\phi_s\bar\rho_s
	=
	\phi_{s0}\bar\rho_{s0}+\mc C_s,
	\qquad
	\mc C_s=\sum_\alpha\mc C_s^\alpha .
	\label{eq:MC_solid_phase_material_mass}
\end{equation}
%
With true local conservation in the solid material,
%
\begin{equation}
	\bar J_s\bar\rho_s=\bar\rho_{s0},
	\label{eq:MC_solid_true_mass_conservation}
\end{equation}
%
\eqref{eq:MC_solid_phase_material_mass} becomes
%
\begin{equation}
	a_s\phi_s\bar\rho_{s0}
	=
	\phi_{s0}\bar\rho_{s0}+\mc C_s .
	\label{eq:MC_solid_distension_mass_relation}
\end{equation}
%
The rate form follows from \(J=a_s\bar J_s\),
%
\begin{empheq}[box=\fbox]{equation}
	\frac{\dot a_s}{a_s}
	=
	\nabla_{\mbf x}\cdot\mbf v_s
	+\frac{\dot{\bar\rho}_s}{\bar\rho_s}.
	\label{eq:MC_solid_distension_evolution}
\end{empheq}
%
For the held-fixed derivatives used next, let
\(\mc X_s=\{\bar{\mbf F}_s^e,\mbf A_s^e,\bar\rho_s,
\eta_s^\beta,\theta\}\).  The intrinsic density rate used in the
Coleman--Noll collection is obtained
from \eqref{eq:MC_solid_true_split}, the isochoric plastic flow, and the
composition-dependent stress-free true deformation,
%
\begin{align}
	\frac{\dot{\bar\rho}_s}{\bar\rho_s}
	={} &
	-\operatorname{tr}
	\left[
		\dot{\bar{\mbf F}}_s^e
		(\bar{\mbf F}_s^e)^{-1}
		\right]
	\notag  \\
	    & -
	\operatorname{tr}\left[
		\left(
		\left.
		\p{\bar{\mbf F}_s^0}{\theta}
		\right\vert_{\mc X_s}
		\dot\theta
		+
		\sum_\beta
		\left.
		\p{\bar{\mbf F}_s^0}{\eta_s^\beta}
		\right\vert_{\mc X_s}
		\dot\eta_s^\beta
		\right)
		(\bar{\mbf F}_s^0)^{-1}
		\right].
	\label{eq:MC_solid_density_rate}
\end{align}
%

The generic fluid and multicomponent solid Helmholtz energies are
%
\begin{equation}
	\begin{alignedat}{2}
		\psi_f
		& =
		\psi_f(
		\bar\rho_f,\eta_f^\beta,\theta),
		\qquad&
		f\in\mc F,
		\\
		\psi_s
		& =
		\psi_s(
		\bar{\mbf F}_s^e,\mbf A_s^e,
		\bar\rho_s,\eta_s^\beta,\theta),
		\qquad&
		s\in\mc S .
	\end{alignedat}
	\label{eq:MC_solid_general_free_energy}
\end{equation}
%
with state sets
%
\begin{equation}
	\mc X_s
	=
	\{
	\bar{\mbf F}_s^e,\mbf A_s^e,
	\bar\rho_s,\eta_s^\beta,\theta
	\},
	\qquad
	\mc X_f
	=
	\{
	\bar\rho_f,\eta_f^\beta,\theta
	\}.
	\label{eq:MC_state_sets}
\end{equation}
%
The notation
\(\left.\partial(\cdot)/\partial z\right\vert_{\mc X_s}\) or
\(\left.\partial(\cdot)/\partial z\right\vert_{\mc X_f}\) means that all other
members of the indicated state set are held fixed.  The composition variables
are constrained by their mass-fraction normalizations, so composition
coefficients are projected onto the corresponding tangent spaces by
subtracting the \(N\)th component coefficient from the \(\beta=1,\ldots,N-1\)
coefficients.

Electrical effects enter the free-energy structure through the phase
current-volume electric enthalpies \(\omega^{+}_{\xi}\) introduced in
\sref{virtual_power_derivation}.  The electric field is a single spatial field,
but each phase may have its own dielectric response to that field, in the same
sense that each phase has its own Helmholtz free energy \(\psi_\xi\).  The
derivative of \(\omega^{+}_{\xi}\) with respect to \(\mbf E\) defines the
phase displacement \(\mbf d_\xi\), and the total displacement is
\(\mbf d=\sum_\xi\phi_\xi\mbf d_\xi\).  Stationarity with respect to
\(\varphi\) gives Gauss' law.  Derivatives with respect to deformation,
porosity, saturation, composition, and temperature add reversible Maxwell and
dielectric contributions to the corresponding Coleman--Noll restrictions.
These reversible terms are not dissipative.  Independent polarization,
dielectric relaxation, and switching are omitted in the present reservoir-scale
model.  Their induced dielectric effect can still be represented, when needed,
by the constitutive choice of \(\omega^{+}_{\xi}\), or equivalently by a
permittivity closure for \(\mbf d_\xi\).
In the collection below, the nonelectrical phase Helmholtz derivatives are
written explicitly, while the electric-potential coefficient is displayed as a
separate electrical term.  The mechanical Maxwell-type contribution in
\eqref{eq:vp_electrical_mechanical_power} is assigned to each phase momentum
coefficient by the volume-fraction-weighted pullback and is included in the
combined reversible stress \(\bs\sigma_\xi^{\prime+}\).  If
\(\omega^{+}_{\xi}\) is not eliminated by a dielectric closure and depends on a
mechanical, compositional, or thermal state variable, its partial derivative is
added to the matching reversible coefficient before imposing the corresponding
Coleman--Noll restriction.

No component additivity has been imposed.  Consequently, the generic
Coleman--Noll argument uses only the phase internal energy \(e_\xi\),
Helmholtz free energy \(\psi_\xi\), and entropy \(\mathfrak{s}_\xi\).  A
component resolution of the solid Helmholtz energy is introduced only in the
additive specialization of \sref{MC_additive_component_specialization}.

\subsection{Coleman--Noll procedure}
\label{sec:MC_coleman_noll_procedure}

The Coleman--Noll procedure is the standard constitutive exploitation of the
local entropy inequality: the constitutive state dependence and balance laws are
substituted into the inequality, coefficients of independent admissible rates
are collected, and the remaining residual terms are required to be dissipative
\cite{coleman1974thermodynamics}.  Let \(e_\xi\) denote the specific internal
energy of phase \(\xi\), and let
\(\mathfrak{s}_\xi\) denote its specific entropy.  For fluid phases,
%
\begin{equation}
	e_f=\psi_f+\theta\mathfrak{s}_f,
	\qquad
	\psi_f=\psi_f(\bar\rho_f,\eta_f^\beta,\theta).
	\label{eq:MC_fluid_legendre_relation}
\end{equation}
%
For multicomponent solid phases,
%
\begin{equation}
	e_s=\psi_s+\theta\mathfrak{s}_s,
	\qquad
	\psi_s=\psi_s(
	\bar{\mbf F}_s^e,\mbf A_s^e,
	\bar\rho_s,\eta_s^\beta,\theta).
	\label{eq:MC_solid_legendre_relation}
\end{equation}
%
The free-energy rates are
%
\begin{align}
	\sum_{f\in\mc F}\rho_f\dot\psi_f
	={} &
	\sum_{f\in\mc F}
	\rho_f
	\left(
	\left.
	\p{\psi_f}{\bar\rho_f}
	\right\vert_{\mc X_f}
	\dot{\bar\rho}_f
	+
	\sum_\beta
	\left.
	\p{\psi_f}{\eta_f^\beta}
	\right\vert_{\mc X_f}
	\dot\eta_f^\beta
	+
	\left.
	\p{\psi_f}{\theta}
	\right\vert_{\mc X_f}
	\dot\theta
	\right),
	\label{eq:MC_fluid_free_energy_rate}
	\\
	\sum_{s\in\mc S}\rho_s\dot\psi_s
	={} &
	\sum_{s\in\mc S}
	\rho_s
	\left[
		\left.
		\p{\psi_s}{\bar{\mbf F}_s^e}
		\right\vert_{\mc X_s}
		:
		\dot{\bar{\mbf F}}_s^e
		+
		\left.
		\p{\psi_s}{\mbf A_s^e}
		\right\vert_{\mc X_s}
		:
		\dot{\mbf A}_s^e
		\right.
	\notag       \\
	    & \left.
		+
		\left.
		\p{\psi_s}{\bar\rho_s}
		\right\vert_{\mc X_s}
		\dot{\bar\rho}_s
		+
		\sum_\beta
		\left.
		\p{\psi_s}{\eta_s^\beta}
		\right\vert_{\mc X_s}
		\dot\eta_s^\beta
		+
		\left.
		\p{\psi_s}{\theta}
		\right\vert_{\mc X_s}
		\dot\theta
		\right].
	\label{eq:MC_solid_free_energy_rate}
\end{align}
%
The fluid phases carry no elastic stress,
%
\begin{equation}
	\bs\sigma_f'=\mbf 0,
	\qquad
	f\in\mc F .
	\label{eq:MC_fluid_stress_free}
\end{equation}
%
When electrical effects are retained, their reversible mechanical stress is
therefore purely Maxwellian:
%
\begin{equation}
	\bs\sigma_f^{\prime+}
	=
	\bs{\sigma}^{+}_{f},
	\qquad
	f\in\mc F .
	\label{eq:MC_fluid_total_reversible_stress}
\end{equation}
%
The interfacial Helmholtz contribution retains the form introduced in
\eqref{eq:interfacial_energy_density}.  Allowing \(\gamma\) to depend on temperature is
consistent with standard capillary-pressure scaling used in reservoir
engineering \cite{leverett1941capillary},
%
\begin{equation}
	\psi_\gamma
	=
	\phi\gamma(\{S_f\}_{f\in\mc F},\theta),
	\qquad
	\dot\psi_\gamma
	=
	\sum_{f\in\mc F}\gamma_f\dot\phi_f
	+
	\phi
	\left.
	\p{\gamma}{\theta}
	\right\vert_{\{S_f\}_{f\in\mc F}}
	\dot\theta .
	\label{eq:MC_interfacial_helmholtz_rate}
\end{equation}
%

Newly converted or relatively transported component mass is assigned the
specific internal energy \(e_\xi\) of the phase receiving or losing that mass.
Let \(\mbf L_\xi=\nabla_{\mbf x}\mbf v_\xi\) be the phase
velocity gradient, \(\mbf q_\xi\) the heat flux, \(\rho_\xi r_\xi\) the external
heat supply per unit current mixture volume, and \(\rho_\xi e_\xi^{\rm int}\)
the internal energy supply from interphase exchange.  Using the total
current-volume component mass-addition rate
\(\dot m_\xi^\alpha=\dot c_\xi^\alpha+\dot{\mc D}_\xi^\alpha/J_\xi\) and the
chemical-potential multiplier convention, the phase energy balance is
%
\begin{align}
	\rho_\xi\dot e_\xi
	+
	\sum_\alpha
	\dot m_\xi^\alpha
	e_\xi
	={} &
	\bs\sigma_\xi^{\prime+}:\mbf L_\xi
	+
	\phi_\xi\bar p_\xi
	\frac{\dot{\bar\rho}_\xi}{\bar\rho_\xi}
	+
	\bar p_\xi\dot\phi_\xi
	+
	\sum_\alpha
	\frac{\pi_\xi^\alpha}{\eta_\xi^\alpha}
	\dot\eta_\xi^\alpha
	\notag  \\
	    & -
	\sum_\alpha
	\left(
	\mu_\xi^\alpha-\psi_\xi
	\right)\dot m_\xi^\alpha
	-
	\nabla_{\mbf x}\cdot\mbf q_\xi
	+
	\rho_\xi r_\xi
	+
	\rho_\xi e_\xi^{\rm int},
	\label{eq:MC_energy_balance}
\end{align}
%
where \(e_\xi=\psi_\xi+\theta\mathfrak{s}_\xi\).
The solid stress power is decomposed by
\(\mbf F=\mbf A_s\bar{\mbf F}_s\),
%
\begin{equation}
	\mbf L_s
	=
	\dot{\mbf A}_s\mbf A_s^{-1}
	+
	\mbf A_s
	\dot{\bar{\mbf F}}_s
	\bar{\mbf F}_s^{-1}
	\mbf A_s^{-1}.
	\label{eq:MC_F_split_for_power}
\end{equation}
%
Using \eqref{eq:MC_solid_kinematic_splits}, the solid stress power is
%
\begin{align}
	\sum_{s\in\mc S}\bs\sigma_s^{\prime+}:\mbf L_s
	={} &
	\sum_{s\in\mc S}
	\operatorname{tr}\left[
		(\bs\sigma_s^{\prime+})^T
		\dot{\mbf A}_s^e
		(\mbf A_s^e)^{-1}
		\right]
	\notag  \\
	    & +
	\sum_{s\in\mc S}
	\operatorname{tr}\left[
		(\mbf A_s^e)^{-1}
		(\bs\sigma_s^{\prime+})^T
		\mbf A_s^e
		\dot{\mbf A}_s^p
		(\mbf A_s^p)^{-1}
		\right]
	\notag  \\
	    & +
	\sum_{s\in\mc S}
	\operatorname{tr}\left[
		(\mbf A_s^p)^{-1}
		(\mbf A_s^e)^{-1}
		(\bs\sigma_s^{\prime+})^T
		\mbf A_s^e\mbf A_s^p
		\left.
		\p{\mbf A_s^0}{\theta}
		\right\vert_{\mc X_s}
		(\mbf A_s^0)^{-1}
		\right]\dot\theta
	\notag  \\
	    & +
	\sum_{s\in\mc S}
	\sum_\beta
	\operatorname{tr}\left[
		(\mbf A_s^p)^{-1}
		(\mbf A_s^e)^{-1}
		(\bs\sigma_s^{\prime+})^T
		\mbf A_s^e\mbf A_s^p
		\left.
		\p{\mbf A_s^0}{\eta_s^\beta}
		\right\vert_{\mc X_s}
		(\mbf A_s^0)^{-1}
		\right]\dot\eta_s^\beta
	\notag  \\
	    & +
	\sum_{s\in\mc S}
	\operatorname{tr}\left[
	\mbf A_s^{-1}
	(\bs\sigma_s^{\prime+})^T
	\mbf A_s
	\dot{\bar{\mbf F}}_s^e
	(\bar{\mbf F}_s^e)^{-1}
	\right]
	\notag  \\
	    & +
	\sum_{s\in\mc S}
	\operatorname{tr}\left[
	(\bar{\mbf F}_s^e)^{-1}
	\mbf A_s^{-1}
	(\bs\sigma_s^{\prime+})^T
	\mbf A_s
	\bar{\mbf F}_s^e
	\dot{\bar{\mbf F}}_s^p
	(\bar{\mbf F}_s^p)^{-1}
	\right]
	\notag  \\
	    & +
	\sum_{s\in\mc S}
	\operatorname{tr}\left[
	(\bar{\mbf F}_s^p)^{-1}
	(\bar{\mbf F}_s^e)^{-1}
	\mbf A_s^{-1}
	(\bs\sigma_s^{\prime+})^T
	\mbf A_s
	\bar{\mbf F}_s^e
	\bar{\mbf F}_s^p
	\left.
	\p{\bar{\mbf F}_s^0}{\theta}
	\right\vert_{\mc X_s}
	(\bar{\mbf F}_s^0)^{-1}
	\right]\dot\theta
	\notag  \\
	    & +
	\sum_{s\in\mc S}
	\sum_\beta
	\operatorname{tr}\left[
	(\bar{\mbf F}_s^p)^{-1}
	(\bar{\mbf F}_s^e)^{-1}
	\mbf A_s^{-1}
	(\bs\sigma_s^{\prime+})^T
	\mbf A_s
	\bar{\mbf F}_s^e
	\bar{\mbf F}_s^p
	\left.
	\p{\bar{\mbf F}_s^0}{\eta_s^\beta}
	\right\vert_{\mc X_s}
	(\bar{\mbf F}_s^0)^{-1}
	\right]\dot\eta_s^\beta .
	\label{eq:MC_stress_power_split}
\end{align}
%

Following the standard Liu multiplier exploitation of constrained entropy
inequalities \cite{liu1972method}, the entropy inequality is
%
\begin{align}
	0\le{} &
	\sum_\xi
	\left[
		\rho_\xi\dot{\mathfrak{s}}_\xi
		+
		\sum_\alpha
		\dot m_\xi^\alpha
		\mathfrak{s}_\xi
		+
		\nabla_{\mbf x}\cdot
		\left(
		\frac{\mbf q_\xi}{\theta}
		\right)
		-
		\frac{\rho_\xi r_\xi}{\theta}
		\right].
	\label{eq:MC_entropy_inequality}
\end{align}
%
Eliminating \(\rho_\xi r_\xi\) with \eqref{eq:MC_energy_balance}, adding the
identically vanishing mass-fraction tangent terms
\(\Pi_\xi\sum_\alpha\dot\eta_\xi^\alpha\), and adding
\(\lambda\sum_\xi\dot\phi_\xi\) together with the reversible electrical
enthalpy terms, gives
%
\begin{align}
	0\le{} &
	\sum_\xi
	\begin{aligned}[t]
		 & \frac{1}{\theta}
		\left(
			\bs\sigma_\xi^{\prime+}:\mbf L_\xi
		+
		\phi_\xi\bar p_\xi
		\frac{\dot{\bar\rho}_\xi}{\bar\rho_\xi}
		-
		\rho_\xi
		\left(
		\dot\psi_\xi+\mathfrak{s}_\xi\dot\theta
		\right)
		-
		\sum_\alpha
		\dot m_\xi^\alpha
		\mu_\xi^\alpha
		+
		\rho_\xi e_\xi^{\rm int}
		\right)
		-
		\frac{\mbf q_\xi\cdot\nabla_{\mbf x}\theta}{\theta^2}
	\end{aligned}
	\notag     \\
	       & +
	\frac{1}{\theta}
	\sum_\xi
	\left(
	\bar p_\xi\dot\phi_\xi
	+
	\sum_\alpha
	\frac{\pi_\xi^\alpha}{\eta_\xi^\alpha}
	\dot\eta_\xi^\alpha
	+
	\Pi_\xi\sum_\alpha\dot\eta_\xi^\alpha
	\right)
	\notag     \\
	       & +
	\frac{1}{\theta}
	\left(
	\lambda\sum_\xi\dot\phi_\xi
		-
		\sum_{f\in\mc F}\gamma_f\dot\phi_f
	\right)
	\notag     \\
	       & +
	\frac{1}{\theta}
	\left[
	-
	\sum_\xi
	\left(
	\phi_\xi\dot\omega^{+}_{\xi}
	+
	\omega^{+}_{\xi}\dot\phi_\xi
	\right)
	+
	\left(
	\varrho-\nabla_{\mbf x}\cdot\mbf d

\subsection{Reversible restrictions}
\label{sec:MC_reversible_restrictions}

The mechanical power in the entropy inequality is written with
\(\bs\sigma_\xi^{\prime+}\).  The restrictions below separate the
Maxwell contribution defined in \eqref{eq:vp_electrical_stress_definition} from
the nonelectrical Helmholtz part \(\bs\sigma_\xi'\); the two are recombined by
\eqref{eq:vp_total_reversible_stress_definition}.

The coefficient of
\(\dot{\bar{\mbf F}}_s^e(\bar{\mbf F}_s^e)^{-1}\) gives
%
\begin{align}

	\mbf A_s
	={} &

	\notag \\

	\notag \\

	\rho_s

	\p{\psi_s}{\bar\rho_s}

	\bar\rho_s

	\right\vert_{\mc X_s,\mbf E}
	\bar\rho_s
	-
	\phi_s\bar p_s
	\right)\mbf I .
	\label{eq:MC_stress_before_pressure}
\end{align}

The density coefficients identify the phase pressures,
%

	\end{align}
\end{subequations}

\end{equation}

\end{equation}

\(\mbf E\otimes\mbf d-\frac{1}{2}\left(\mbf E\cdot\mbf d\right)\mbf I\), with

\subsection{Sources, residual dissipation, and closures}
\label{sec:MC_residual_dissipation}

The total current-volume component mass-addition rate separates into conversion
and relative-transport accumulation,
\(\dot m_\xi^\alpha=\dot c_\xi^\alpha+\dot{\mc D}_\xi^\alpha/J_\xi\).  Therefore
the source contribution in \eqref{eq:MC_reduced_entropy_inequality} splits as
%
\begin{align}
	-\sum_\xi\sum_\alpha
	\dot m_\xi^\alpha\mu_\xi^\alpha
	={} &
	-\sum_\xi\sum_\alpha
	\dot c_\xi^\alpha\mu_\xi^\alpha
	-
	\sum_\xi\sum_\alpha
	\frac{\dot{\mc D}_\xi^\alpha}{J_\xi}\mu_\xi^\alpha
	\notag \\
	={} &
	-\sum_m
	\left[
		\sum_\xi\sum_\alpha
		\mu_\xi^\alpha
		\nu_{\xi (m)}^\alpha
		\right]\dot r_{(m)}
	-
	\sum_\xi\sum_\alpha
	\frac{\dot{\mc D}_\xi^\alpha}{J_\xi}\mu_\xi^\alpha .
	\label{eq:MC_helmholtz_source_projection}
\end{align}
%
The last term is the transport-accumulation contribution,
%
\begin{equation}
	-\sum_\xi\sum_\alpha
	\frac{\dot{\mc D}_\xi^\alpha}{J_\xi}
	\mu_\xi^\alpha .
	\label{eq:MC_helmholtz_transport_accumulation}
\end{equation}
%

%
\begin{align*}

	\frac{\dot{\mc D}_\xi^\alpha}{J_\xi}
	\mu_\xi^\alpha
	={} &
	\sum_\xi\sum_\alpha
	\nabla_{\mbf x}\cdot
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)
	\mu_\xi^\alpha
	\notag  \\
	={} &
	\sum_\xi\sum_\alpha
	\nabla_{\mbf x}\cdot
	\left[
		\mu_\xi^\alpha
		\left(
		\mbf j_{\xi,{\rm disp}}^\alpha
		+
		\mbf j_{\xi,{\rm diff}}^\alpha
		\right)
		\right]
	\notag  \\
	    & -
	\sum_\xi\sum_\alpha
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)\cdot\nabla_{\mbf x}\mu_\xi^\alpha .
\end{align*}
%
Equivalently, over an arbitrary current region, integration by parts gives the
boundary term plus the local transport power,
%
\begin{align*}
	\int_{\mc B}
	\sum_\xi\sum_\alpha
	\nabla_{\mbf x}\cdot
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)
	\mu_\xi^\alpha\dv
	={} &
	\int_{\partial\mc B}
	\sum_\xi\sum_\alpha
	\mu_\xi^\alpha
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)\cdot\mbf n\,{\rm d}a
	\notag  \\
	    & -
	\int_{\mc B}
	\sum_\xi\sum_\alpha
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)\cdot\nabla_{\mbf x}\mu_\xi^\alpha\dv .
\end{align*}
%
The boundary contribution belongs to chemical-energy transport through
\(\partial\mc B\), not local entropy production.  The local transport power is
therefore
%
\begin{equation}
	-\sum_\xi\sum_\alpha
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)\cdot\nabla_{\mbf x}\mu_\xi^\alpha.
	\label{eq:MC_relative_transport_power}
\end{equation}
%
This is the standard species-flux contribution in continuum thermodynamics of
reacting mixtures~\cite{bothe2015continuum,gray2009tcat_transport}.

With electrical effects, the charge-projected constraint \(C_5\)
\eqref{eq:vp_C5_charge_constraint} contains the same component-rate bracket as
the chemomechanical constraint \(C_4\).  The variational derivation therefore
combines the two before varying them, writes the chemomechanical multiplier as
\(\hat{\mu}_\xi^\alpha\), and uses the electric potential \(\varphi\) directly
as the multiplier conjugate to the charge projection.  The component-rate
multiplier used below is therefore the electrochemical potential
%
\begin{equation}
	\mu_\xi^\alpha
	=
	\hat{\mu}_\xi^\alpha
	+
	z_\xi^\alpha\varphi .
	\label{eq:MC_electrochemical_multiplier_split}
\end{equation}
%
The factor multiplying \(\varphi\) is the component charge per unit mass
\(z_\xi^\alpha\), not the mixture charge density \(\varrho\).  With this
identification, the local conversion and transport powers retain their original
form,
%
\begin{align}
	-\sum_\xi\sum_\alpha
	\dot c_\xi^\alpha\mu_\xi^\alpha
	&=
	\sum_m
	\mc A_{(m)}\dot r_{(m)},
	\qquad
	\mc A_{(m)}
	=
	-\sum_\xi\sum_\alpha
	\mu_\xi^\alpha\nu_{\xi (m)}^\alpha,
	\label{eq:MC_electrochemical_affinity_projection}
	\\
	-\sum_\xi\sum_\alpha
	\frac{\dot{\mc D}_\xi^\alpha}{J_\xi}
	\mu_\xi^\alpha
	&=
	-\sum_\xi\sum_\alpha
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)\cdot\nabla_{\mbf x}\mu_\xi^\alpha
	+
	\text{b.t.}
	\label{eq:MC_electrochemical_relative_transport_power}
\end{align}
%
For a local mechanism satisfying
\eqref{eq:mechanism_charge_conservation}, the explicit electric-potential part
of the decomposed affinity cancels:
%
\begin{equation*}
	-\sum_\xi\sum_\alpha
	\mu_\xi^\alpha\nu_{\xi (m)}^\alpha
	=
	-\sum_\xi\sum_\alpha
	\hat{\mu}_\xi^\alpha\nu_{\xi (m)}^\alpha
	-
	\varphi
	\sum_\xi\sum_\alpha
	z_\xi^\alpha\nu_{\xi (m)}^\alpha
	=
	-\sum_\xi\sum_\alpha
	\hat{\mu}_\xi^\alpha\nu_{\xi (m)}^\alpha .
\end{equation*}
%
Ion exchange enters the residual
inequality through the same mechanism-rate power as other reactions, while the
electric field affects the exchange problem through electrochemical transport,
Gauss' law, and the state dependence of the free energy.

Converted and relatively transported mass of the same component enter the phase
material measure only through their sum.  The variational derivation leaves the
generalized work potential \(L_\xi^\alpha\) explicit in
\eqref{eq:el_conversion_component}.  We make the admissible choice
%
\begin{equation}
	L_\xi^\alpha
	=
	\mu_\xi^\alpha-\psi_\xi .
	\label{eq:MC_L_mu_psi_closure}
\end{equation}
%
so that \(\psi_\xi+L_\xi^\alpha=\mu_\xi^\alpha\) for each phase/component
source.
The mechanism affinity is
%
\begin{equation}
	\mc A_{(m)}
	=
	-
	\sum_\xi\sum_\alpha
	\mu_\xi^\alpha\nu_{\xi (m)}^\alpha .
	\label{eq:MC_affinity_projection}
\end{equation}
%
Substituting \eqref{eq:MC_L_mu_psi_closure} into the multiplier evolution
equation \eqref{eq:el_conversion_component} gives the
chemical-potential-multiplier evolution law
%
\begin{empheq}[box=\fbox]{equation}
	\frac{D_\xi\mu_\xi^\alpha}{Dt}
	=
	\frac{1}{2}
	\mbf v_\xi\cdot\mbf v_\xi
	-
	\mu_\xi^\alpha
	-
	\frac{\pi_\xi^\alpha}{\phi_\xi\bar\rho_\xi\eta_\xi^\alpha}.
	\label{eq:MC_admissible_conversion_component}
\end{empheq}
%
Equation \eqref{eq:MC_admissible_conversion_component} follows from the
accumulated-coordinate variations of the conversion and transport measures in
the constrained Hamilton--d'Alembert statement because both enter the component
material measure through their sum.  Thus they share the single multiplier
\(\mu_\xi^\alpha\), and the admissible choice
\eqref{eq:MC_L_mu_psi_closure} makes the corresponding transfer potential
consistent with the thermodynamic chemical potential.  Equation
\eqref{eq:MC_admissible_conversion_component}, together with an initial value,
determines \(\mu_\xi^\alpha\) along the motion of phase \(\xi\).
%
\subsubsection{Plastic flow rules and residual dissipation}
The plastic-deformation power in the residual inequality is driven by the
combined reversible stress of each solid phase mapped through that phase's
distension and true-deformation gradients to its plastic-flow configuration.
Because each true-plastic flow is isochoric, only the deviatoric part of that
mapped stress contributes to the true-plastic power.  Define \(\mbf S_s^{+}\) as
the deviatoric true-plastic driving stress,
%
\begin{equation*}
	\mbf S_s^{+}
	\equiv
	(\bar{\mbf F}_s^e)^{-1}
	\mbf A_s^{-1}
	\left(
	(\bs\sigma_s^{\prime+})^T
	-\frac{1}{3}
	\operatorname{tr}\left(\bs\sigma_s^{\prime+}\right)\mbf I
	\right)
	\mbf A_s
	\bar{\mbf F}_s^e,
	\qquad
	s\in\mc S.
\end{equation*}
%
The trace-free property follows from the similarity transformation of the
deviatoric stress,
%
\begin{equation*}
	\operatorname{tr}\mbf S_s^{+}
	=0,
	\qquad
	s\in\mc S .
\end{equation*}
%
Thus any spherical part of the mapped true-plastic driving stress does no power
against \(\dot{\bar{\mbf F}}_s^p(\bar{\mbf F}_s^p)^{-1}\).  This is the direct
analogue of Drumheller's \cite[eqs.~(108)--(112)]{drumheller2000}
true-plastic driving stress: his deviatoric true stress is replaced here by the
phase single-prime solid stress first mapped through that phase's distension
tensor.  Starting from \eqref{eq:MC_collected_entropy}, the residual inequality
below follows after substituting the reversible restrictions
	\eqref{eq:MC_phase_pressure_definitions}, \eqref{eq:MC_solid_stress_definition},
		\eqref{eq:MC_distension_force_definition},
		\eqref{eq:MC_volume_fraction_restrictions},
		\eqref{eq:MC_fluid_composition_projection},
		\eqref{eq:MC_solid_composition_projection}, and
		\eqref{eq:MC_entropy_definitions}, together with the Gauss-law restriction
		\eqref{eq:gauss_law}.  When the electrical enthalpy is mechanically active,
		the stress power uses the combined reversible stress
		\(\bs\sigma_\xi^{\prime+}\), so the Maxwell-type term is removed with the
		other reversible mechanical terms.  The source and
		transport reductions are
		\eqref{eq:MC_helmholtz_source_projection},
		\eqref{eq:MC_affinity_projection}, and
		\eqref{eq:MC_relative_transport_power}.  In the electrical theory, the
	corresponding electrochemical reductions are
	\eqref{eq:MC_electrochemical_affinity_projection} and
	\eqref{eq:MC_electrochemical_relative_transport_power}.  With those
	substitutions made, the Gauss-law restriction removes the
	electric-potential-rate term and the remaining inequality can be written as
%
\begin{equation}
	\begin{aligned}
		0\le{}\; &
		\sum_m\mc A_{(m)}\dot r_{(m)}
		-
		\sum_\xi\sum_\alpha
		\left(
		\mbf j_{\xi,{\rm disp}}^\alpha
		+
		\mbf j_{\xi,{\rm diff}}^\alpha
		\right)\cdot
		\nabla_{\mbf x}\mu_\xi^\alpha
		\\
		         &+
		\sum_{s\in\mc S}
		\bs\sigma_s^{\prime+}:
		\mbf A_s^e
		\dot{\mbf A}_s^p
		(\mbf A_s^p)^{-1}
		(\mbf A_s^e)^{-1}
		\\
		         & +
		\sum_{s\in\mc S}
		\operatorname{tr}\left[
			\mbf S_s^{+}
			\dot{\bar{\mbf F}}_s^p
			(\bar{\mbf F}_s^p)^{-1}
			\right]
		\\
		         & +
		\sum_\xi\rho_\xi e_\xi^{\rm int}
		-
		\sum_\xi
		\frac{\mbf q_\xi\cdot\nabla_{\mbf x}\theta}{\theta}.
	\end{aligned}
	\label{eq:MC_residual_dissipation}
\end{equation}
%
\subsubsection{Constitutive restrictions for the process-rate forces}

The Hamilton--d'Alembert statement varies the conservative coordinates, and the
accompanying Onsager stationarity condition varies the conversion and
relative-transport rates.  The entropy inequality then restricts the
constitutive relation between those rates and their forces.  The variational
force balances \eqref{eq:el_conversion_affinity} and
\eqref{eq:hamilton_dispersion_force_balance} are already written in terms of the
chemical-potential multiplier \(\mu_\xi^\alpha\) identified by the entropy
inequality.  In the electrical theory this multiplier is electrochemical.  The
relative component fluxes redistribute mass among the components of a phase;
they do not move the whole phase mass together.  Adding the same component
potential to every component in a phase therefore does not change the transport
power.  The driving quantities are component-potential differences, with the
neutral theory recovered by setting \(z_\xi^\alpha=0\).

Substitution of the entropy-identified chemical force into the transport term of
\eqref{eq:MC_residual_dissipation} gives
%
\begin{align}
	-   & \sum_\xi\sum_\alpha
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)\cdot\nabla_{\mbf x}\mu_\xi^\alpha
	\notag                    \\
	={} &
	\sum_\xi\sum_\alpha
	\left(
	\mbf M_{\xi,{\rm disp}}^\alpha\cdot
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf M_{\xi,{\rm diff}}^\alpha\cdot
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right).
	\label{eq:MC_relative_transport_force_rate_power}
\end{align}
%
Thus \(\mc A_{(m)}\) is conjugate to \(\dot r_{(m)}\), while each
\(\mbf M_{\xi,k}^\alpha\) is conjugate to its relative flux
\(\mbf j_{\xi,k}^\alpha\).  Diffusion and dispersion are distinct dissipative
mechanisms with potentially different resistance operators, even though both
respond to the same component-potential gradient.

The variational equations alone do not prescribe how the dissipative forces
depend on their rates.  The residual inequality
\eqref{eq:MC_residual_dissipation} requires
\(\sum_m\mc A_{(m)}\dot r_{(m)}\ge0\) and
\(\sum_{\xi,\alpha}\mbf M_{\xi,k}^\alpha\cdot
\mbf j_{\xi,k}^\alpha\ge0\) for an uncoupled sufficient construction.  Near
equilibrium, a quadratic dissipation potential is the standard linear Onsager
constitutive choice~\cite{doi2011onsager,wang2021onsager}.  For these rates,
choose
%
\begin{align}
	\Phi_{\rm rt}
	={} &
	\frac{1}{2}\int_{\mc B}
	\sum_\xi\sum_\alpha\sum_\beta
	\mbf j_{\xi,{\rm disp}}^\alpha
	\cdot\mbf R_{\xi,{\rm disp}}^{\alpha\beta}
	\mbf j_{\xi,{\rm disp}}^\beta\dv
	\notag  \\
	    & +
	\frac{1}{2}\int_{\mc B}
	\sum_\xi\sum_\alpha\sum_\beta
	\mbf j_{\xi,{\rm diff}}^\alpha
	\cdot\mbf R_{\xi,{\rm diff}}^{\alpha\beta}
	\mbf j_{\xi,{\rm diff}}^\beta\dv
	\notag  \\
	    & +
	\frac{1}{2}\int_{\mc B}
	\sum_m\frac{\dot r_{(m)}^2}{K_{(m)}}\dv,
	\label{eq:MC_reaction_transport_dissipation_potential}
\end{align}
%
where \(K_{(m)}>0\), and the two resistance tensors are symmetric positive
semidefinite for component fluxes satisfying
\eqref{eq:averaged_relative_flux_sum}.  Their positivity, and the omission of
cross-coupling between the two rates, are constitutive assumptions rather than
consequences of averaging~\cite{doi2011onsager,gray2009tcat_transport}.
Differentiating the potential with respect to the mechanism rate gives the
constitutive relation
%
\begin{equation}
	\mc A_{(m)}
	=
	\frac{\partial\Phi_{\rm rt}}{\partial\dot r_{(m)}}
	=
	\frac{\dot r_{(m)}}{K_{(m)}},
	\qquad\text{hence}\qquad
	\dot r_{(m)}=K_{(m)}\mc A_{(m)}.
	\label{eq:MC_onsager_reaction_rate}
\end{equation}
%
Likewise, the constitutive dissipative forces are the flux derivatives of the
same potential.  Combining them with the entropy-identified chemical forces,
which agree with the variational multiplier forces after enforcing the
component-flux sum constraint, gives
%
\begin{align}
	\mbf M_{\xi,{\rm disp}}^\alpha
	 & =
	\sum_\beta\mbf R_{\xi,{\rm disp}}^{\alpha\beta}
	\mbf j_{\xi,{\rm disp}}^\beta,
	\notag      \\
	\mbf M_{\xi,{\rm disp}}^\alpha
	+\nabla_{\mbf x}\mu_\xi^\alpha
	 & =\mbf 0,
	\label{eq:MC_onsager_dispersion_stationarity}
	\\
	\mbf M_{\xi,{\rm diff}}^\alpha
	 & =
	\sum_\beta\mbf R_{\xi,{\rm diff}}^{\alpha\beta}
	\mbf j_{\xi,{\rm diff}}^\beta,
	\notag      \\
	\mbf M_{\xi,{\rm diff}}^\alpha
	+\nabla_{\mbf x}\mu_\xi^\alpha
	 & =\mbf 0.
	\label{eq:MC_onsager_diffusion_stationarity}
\end{align}
%
Equivalently, invert the resistance tensor for relative fluxes satisfying
\eqref{eq:averaged_relative_flux_sum} and define
%
\begin{equation*}
	\mbf D_{\xi,k}^{\alpha\beta}
	\equiv
	\left(\mbf R_{\xi,k}^{-1}\right)^{\alpha\beta},
	\qquad
	k\in\{{\rm disp},{\rm diff}\}.
\end{equation*}
%
The diffusion and dispersion closures are
%
\begin{align}
	\mbf j_{\xi,{\rm disp}}^\alpha
	 & =-\sum_\beta
	\mbf D_{\xi,{\rm disp}}^{\alpha\beta}
	\nabla_{\mbf x}\mu_\xi^\beta,
	\label{eq:MC_dispersion_closure}
	\\
	\mbf j_{\xi,{\rm diff}}^\alpha
	 & =-\sum_\beta
	\mbf D_{\xi,{\rm diff}}^{\alpha\beta}
	\nabla_{\mbf x}\mu_\xi^\beta,
	\label{eq:MC_diffusion_closure}
	\\
	\sum_\alpha\mbf D_{\xi,k}^{\alpha\beta}
	 & =\sum_\beta\mbf D_{\xi,k}^{\alpha\beta}=\mbf 0,
	\qquad k\in\{{\rm disp},{\rm diff}\}.
	\label{eq:MC_relative_transport_closures}
\end{align}
%
The resulting constitutively admissible rates satisfy
%
\begin{align}
	 & \sum_m\mc A_{(m)}\dot r_{(m)}
	-
	\sum_\xi\sum_\alpha
	\left(
	\mbf j_{\xi,{\rm disp}}^\alpha
	+
	\mbf j_{\xi,{\rm diff}}^\alpha
	\right)\cdot\nabla_{\mbf x}\mu_\xi^\alpha
	\notag                           \\
	 & \qquad=
	\sum_m\frac{\dot r_{(m)}^2}{K_{(m)}}
	+
	\sum_\xi\sum_\alpha\sum_\beta
	\mbf j_{\xi,{\rm disp}}^\alpha
	\cdot\mbf R_{\xi,{\rm disp}}^{\alpha\beta}
	\mbf j_{\xi,{\rm disp}}^\beta
	+
	\sum_\xi\sum_\alpha\sum_\beta
	\mbf j_{\xi,{\rm diff}}^\alpha
	\cdot\mbf R_{\xi,{\rm diff}}^{\alpha\beta}
	\mbf j_{\xi,{\rm diff}}^\beta
	\ge0.
	\label{eq:MC_onsager_reaction_transport_dissipation}
\end{align}
%
Averaging gives separate diffusion and dispersion rates, but the entropy
inequality gives both rates the same component-potential driving force.  In the
electrical theory this potential includes electric work; the neutral theory is
recovered by setting \(z_\xi^\alpha=0\).  The resistance
tensors are constitutive choices.  In particular, the dispersion mobility may
depend on resolved flow and pore geometry, and a pore-scale closure problem is
required to determine it
~\cite{plumb1988dispersion,gray2009tcat_transport}.  Cross-coupled resistance
tensors can be added without changing the variational statement when the
material requires them~\cite{doi2011onsager}.

No residual polarization term appears in \eqref{eq:MC_residual_dissipation}
because independent polarization is not included as a state variable.  Models
for induced-polarization measurements, ferroelectric switching, or dielectric
relaxation would enlarge the electrical state and add a corresponding
stationarity or dissipative evolution law; those mechanisms are deliberately
left outside the present reservoir-flow specialization.

Following the same sufficient construction used by
Drumheller~\cite{drumheller2000}, one may choose generic uncoupled plastic flow
rules
%
\begin{subequations}
	\label{eq:MC_generic_plastic_flow_rules}
	\begin{align}
		\dot{\bar{\mbf F}}_s^p(\bar{\mbf F}_s^p)^{-1}
		 & =
		\Lambda_{\bar{\mbf F}_s}(\mc X_s)
		(\mbf S_s^{+})^T,
		\label{eq:MC_generic_plastic_deformation_flow_rule}            \\
		\dot{\mbf A}_s^p(\mbf A_s^p)^{-1}
		 & =
		\Lambda_{\mbf A_s}(\mc X_s)
		(\mbf A_s^e)^T
		\bs\sigma_s^{\prime+}
		(\mbf A_s^e)^{-T},
		\label{eq:MC_generic_plastic_distension_flow_rule}             \\
		\Lambda_{\bar{\mbf F}_s}(\mc X_s)
		 & \ge0,
		\label{eq:MC_generic_plastic_deformation_mobility_nonnegative} \\
		\Lambda_{\mbf A_s}(\mc X_s)
		 & \ge0,
		\qquad
		s\in\mc S .
		\label{eq:MC_generic_plastic_distension_mobility_nonnegative}
	\end{align}
\end{subequations}
%
Then the isochoric constraint is preserved by
\(\operatorname{tr}\mbf S_s^{+}=0\), and the two plastic powers in
\eqref{eq:MC_residual_dissipation} are nonnegative term by term.  Composition
dependence of the plastic closure is admissible at this residual level.  It
does not add a new Coleman--Noll coefficient unless the plastic variables are
eliminated as instantaneous functions of \(\eta_s^\beta\).  With the
internal-variable description, composition-dependent hardening, softening,
yield strength, and mobility enter through flow laws for
\(\dot{\mbf A}_s^p(\mbf A_s^p)^{-1}\) and
\(\dot{\bar{\mbf F}}_s^p(\bar{\mbf F}_s^p)^{-1}\) whose arguments include
\(\eta_s^\beta\).  Component-resolved plastic closures for the additive
specialization are given in
\eqref{eq:MC_component_resolved_plastic_flow_rules}.  More general coupled
plastic deformation--distension laws are allowed if their combined plastic
power is nonnegative.  A stronger model with component-specific plastic deformation
gradients would require a different solid kinematic split and new residual
plastic powers.

This admissibility statement is deliberately weaker than an associated
plasticity postulate.  In classical soil and rock plasticity, associated flow
uses the same function to define both the onset of plastic loading and the
direction of irreversible deformation.  Many geomaterials are modeled
non-associatively because the mechanisms that control shear strength are not
the same mechanisms that control dilatancy, pore collapse, grain rearrangement,
cementation, dissolution, or precipitation.  Imposing associated flow on a
frictional material can therefore predict an unrealistic volumetric response.
The mixture formulation has the same physical separation, but it is
expressed through the distension split rather than through a classical yield
surface and plastic potential.  The elastic stress follows from the
stress-generating part of the free energy, whereas
\(\mbf A_s^p\) describes irreversible pore-volume or solid-packing change.
Its evolution is thermodynamically admissible when its contribution to
\eqref{eq:MC_residual_dissipation} is nonnegative; it need not be normal to the
elastic stress potential, nor to the same scalar loading function used to
calibrate shear yielding.  Thus plastic distension may be driven by hydrostatic
stress, composition, reaction affinity, cementation state, or other internal
variables, provided the resulting coupled deformation--distension power remains
nonnegative.  In this sense the formulation contains the physical content
usually associated with non-associated flow in geomechanics without making
normality a constitutive axiom.

This distinction is important because non-associated plasticity is not
thermodynamically consistent merely because it is useful empirically.  Once the
plastic flow direction is separated from the yield or loading function, the
plastic stress power need not be nonnegative automatically, the usual
maximum-dissipation interpretation may be lost, and the algorithmic tangent is
generally nonsymmetric.  The formulation does not remove all of those
mathematical and numerical consequences.  Rather, it fixes the thermodynamic
accounting problem by displaying the true-plastic and plastic-distension powers
explicitly in \eqref{eq:MC_residual_dissipation}.  A non-associated closure is
admissible only after its chosen rates are tested against those conjugate power
terms.  The uncoupled rules in \eqref{eq:MC_generic_plastic_flow_rules} give one
sufficient construction, while more general coupled rules must satisfy the same
nonnegative combined plastic-power requirement.

It remains to make the interaction and heat-conduction terms in
\eqref{eq:MC_residual_dissipation} nonnegative.  For reacting porous mixtures a
pairwise quadratic construction gives an admissible interphase exchange
\cite{foster2026reactingporous}.  Following Bedford and Drumheller
\cite{drumheller1980thermomechanical}, conservation of total mixture energy
requires the internal interaction-energy supplies and the interphase momentum
supplies to satisfy
%
\begin{equation}
	\sum_\xi
	\left(
	\rho_\xi e_\xi^{\rm int}
	+\mbf b_\xi\cdot\mbf v_\xi
	\right)
	=0 .
	\label{eq:MC_interaction_energy_conservation}
\end{equation}
%
Decompose the net momentum supply into pairwise forces,
%
\begin{equation}
	\mbf b_\xi
	=
	\sum_{\zeta\ne\xi}
	\mbf b_{\zeta\xi},
	\qquad
	\mbf b_{\zeta\xi}
	=
	-\mbf b_{\xi\zeta},
	\label{eq:MC_pairwise_interaction_force}
\end{equation}
%
where \(\mbf b_{\zeta\xi}\) is the force exerted on phase \(\xi\) by phase
\(\zeta\).  Then \eqref{eq:MC_interaction_energy_conservation} gives the
pairwise interaction power
%
\begin{equation}
	\sum_\xi\rho_\xi e_\xi^{\rm int}
	=
	-\sum_{\xi<\zeta}
	\mbf b_{\zeta\xi}\cdot
	\left(
	\mbf v_\xi-\mbf v_\zeta
	\right).
	\label{eq:MC_interaction_power_pairing}
\end{equation}
%
A sufficient dissipative closure is obtained by introducing symmetric positive
definite interphase mobility tensors \(\mbf k_{\xi\zeta}\) and symmetric
positive semidefinite heat-conductivity tensors \(\mbf K_\xi\),
%
\begin{subequations}
	\label{eq:MC_interaction_heat_closures}
	\begin{align}
		\mbf b_{\zeta\xi}
		 & =
		-\mbf k_{\xi\zeta}^{-1}
		(\mbf v_\xi-\mbf v_\zeta),
		\qquad
		\mbf k_{\xi\zeta}=\mbf k_{\xi\zeta}^T>0,
		\label{eq:MC_interphase_exchange_law} \\
		\mbf q_\xi
		 & =
		-\mbf K_\xi\nabla_{\mbf x}\theta,
		\qquad
		\mbf K_\xi=\mbf K_\xi^T\ge0 .
		\label{eq:MC_heat_flux_law}
	\end{align}
\end{subequations}
%
where \(\mbf y\cdot\mbf k_{\xi\zeta}\mbf y>0\) for every
\(\mbf y\ne\mbf 0\), \(\mbf y\cdot\mbf K_\xi\mbf y\ge0\), and \(\theta>0\).
Substitution into \eqref{eq:MC_interaction_power_pairing} gives
%
\begin{equation}
	\sum_\xi\rho_\xi e_\xi^{\rm int}
	=
	\sum_{\xi<\zeta}
	\left(
	\mbf v_\xi-\mbf v_\zeta
	\right)
	\cdot
	\mbf k_{\xi\zeta}^{-1}
	\left(
	\mbf v_\xi-\mbf v_\zeta
	\right)
	\ge0 ,
	\label{eq:MC_interphase_exchange_positive}
\end{equation}
%
and the heat-conduction part becomes
%
\begin{equation}
	-\sum_\xi
	\frac{\mbf q_\xi\cdot\nabla_{\mbf x}\theta}{\theta}
	=
	\sum_\xi
	\frac{
		\nabla_{\mbf x}\theta
		\cdot
		\mbf K_\xi
		\nabla_{\mbf x}\theta
	}{\theta}
	\ge0 .
	\label{eq:MC_heat_flux_positive}
\end{equation}
%
Equations \eqref{eq:MC_onsager_reaction_transport_dissipation},
\eqref{eq:MC_interphase_exchange_positive}, and
\eqref{eq:MC_heat_flux_positive}, together with admissible reaction kinetics
and plastic flow laws, are sufficient to make
\eqref{eq:MC_residual_dissipation} nonnegative.

\subsection{Sorption/desorption mechanisms}
\label{sec:MC_sorption_desorption_mechanisms}

Sorption/desorption is a conversion mechanism that transfers a component
between a mobile phase and a solid phase without introducing a separate solid
motion.  For a component \(g\) moving between mobile phase \(f\) and solid phase
\(s\), the stoichiometric coefficients may be written
%
\begin{equation}
	\nu_{s (m)}^g=1,
	\qquad
	\nu_{f (m)}^g=-1,
	\qquad
	\nu_{\xi (m)}^\alpha=0
	\quad\hbox{otherwise}.
	\label{eq:MC_sorption_stoichiometry}
\end{equation}
%
The affinity is
%
\begin{equation}
	\mc A_{(m)}
	=
	\mu_f^g-\mu_s^g .
	\label{eq:MC_sorption_affinity}
\end{equation}
%
An admissible kinetic law must satisfy
\(\mc A_{(m)}\dot r_{(m)}\ge0\).  The sorbed amount \(\eta_s^g\) can also
enter \(\psi_s\), \(\mbf A_s^0\), \(\bar{\mbf F}_s^0\), and the
plastic flow law, so sorption can change stiffness, yield strength, swelling,
and compaction behavior in one thermodynamic state.

Hydrate formation and dissociation can be represented in the same
sorption/desorption form.  For methane hydrate or another clathrate solid, host
water and guest methane are solid components of the same load-bearing solid
phase, while exchange with mobile components changes the solid composition.
That composition may change the free energy, thermal expansion, stress-free
cage volume, and plastic response without introducing a second solid
displacement field.

\subsection{Phase reaction-modified Biot coefficients}
\label{sec:MC_phase_biot_legendre_transform}

The restrictions in \sref{MC_reversible_restrictions} define the single-prime
solid stress directly from the phase Helmholtz energy.  That stress is paired
with the elastic rates in the entropy inequality when the intrinsic solid
density is fixed.  Poromechanics measurements and pressure-based momentum
balances are usually cleaner at fixed solid pressure.  The corresponding
double-prime stress is obtained by a phase-level Legendre transform in the
standard thermodynamic sense \cite{callen1985thermodynamics}, before any
additive component specialization is introduced.

Define the phase and intrinsic specific volumes
%
\begin{equation}
	v_s=\frac{1}{\rho_s},
	\qquad
	v_{s0}=\frac{1}{\rho_{s0}},
	\qquad
	\bar v_s=\frac{1}{\bar\rho_s},
	\qquad
	\bar v_{s0}=\frac{1}{\bar\rho_{s0}},
	\qquad
	s\in\mc S .
	\label{eq:MC_phase_specific_volumes}
\end{equation}
%
The phase pressure definition \eqref{eq:MC_solid_pressure_definition} is
equivalently
%
\begin{equation}
	\bar p_s
	=
	-
	\left.
	\p{\psi_s}{\bar v_s}
	\right\vert_{\mc X_s},
	\qquad
	s\in\mc S ,
	\label{eq:MC_phase_pressure_specific_volume}
\end{equation}
%
where \(\bar v_s\) is the specific-volume representation of the density
argument \(\bar\rho_s\) in \(\mc X_s\).  The phase Legendre transform is
%
\begin{subequations}
	\label{eq:MC_phase_legendre_transform}
	\begin{gather}
		\widetilde\psi_s
		=
		\psi_s+\bar p_s\bar v_s,
		\label{eq:MC_phase_legendre_transform_definition} \\
		\bar v_s
		=
		\left.
		\p{\widetilde\psi_s}{\bar p_s}
		\right\vert_{\bar{\mbf F}_s^e,\mbf A_s^e,\eta_s^\beta,\theta}.
		\label{eq:MC_phase_legendre_transform_pressure_derivative}
	\end{gather}
\end{subequations}
%
For the fixed-pressure elastic derivative, the density dependence induced by
\(\bar{\mbf F}_s^e\) enters through the true Jacobian \(J=a_s\bar J_s\).  This
follows from the material mass balance \eqref{eq:material_mass_2}, and from its
true-solid specialization \eqref{eq:MC_solid_true_mass_conservation}: density
and specific volume depend on the local volume change, hence on the Jacobian,
rather than on the full deformation tensor.  The current values of the
distension, plastic, stress-free, thermal, and composition variables are part of
the state at which the tangent is evaluated, while the differentiation is taken
at fixed phase pressure, temperature, and composition,
%
\begin{align}
	\left.
	\p{\bar v_s}{\bar{\mbf F}_s^e}
	\right\vert_{\bar p_s,\theta,\eta_s^\beta}
	 & =
	\left.
	\p{\bar v_s}{J}
	\right\vert_{\bar p_s,\theta,\eta_s^\beta}
	\frac{\partial J}{\partial\bar{\mbf F}_s^e}
	\notag \\
	 & =
	\left.
	\p{\bar v_s}{J}
	\right\vert_{\bar p_s,\theta,\eta_s^\beta}
	a_s\bar J_s
	(\bar{\mbf F}_s^e)^{-T}.
	\label{eq:MC_phase_specific_volume_deformation_chain_rule}
\end{align}
%
The phase reference energy per unit skeleton reference volume is
\(\rho_{s0}\psi_s\).  The single-prime and double-prime phase Piola stresses are
defined by
%
\begin{subequations}
	\label{eq:MC_phase_pk_stresses}
	\begin{align}
		\mbf P_s'
		 & =
		\rho_{s0}
		\left.
		\p{\widetilde\psi_s}{\bar{\mbf F}_s^e}
		\right\vert_{\bar p_s,\theta,\eta_s^\beta}
		\label{eq:MC_phase_single_prime_pk_stress} \\
		\rho_{s0}
		\left.
		\p{\widetilde\psi_s}{\bar{\mbf F}_s^e}
		\right\vert_{\bar p_s,\theta,\eta_s^\beta}
		 & =
		\rho_{s0}
		\left.
		\p{\psi_s}{\bar{\mbf F}_s^e}
		\right\vert_{\bar v_s,\eta_s^\beta,\theta},
		\label{eq:MC_phase_legendre_pk_identity}   \\
		\mbf P_s^{\prime\prime}
		 & =
		\rho_{s0}
		\left.
		\p{\psi_s}{\bar{\mbf F}_s^e}
		\right\vert_{\bar p_s,\theta,\eta_s^\beta}.
		\label{eq:MC_phase_double_prime_pk_stress}
	\end{align}
\end{subequations}
%
Using the Legendre transform at fixed \(\bar p_s\),
%
\begin{equation}
	\mbf P_s'
	=
	\mbf P_s^{\prime\prime}
	+
	\bar p_s\rho_{s0}
	\left.
	\p{\bar v_s}{\bar{\mbf F}_s^e}
	\right\vert_{\bar p_s,\theta,\eta_s^\beta}.
	\label{eq:MC_phase_pk_legendre_split}
\end{equation}
%
Substitution of \eqref{eq:MC_phase_specific_volume_deformation_chain_rule}
identifies the phase nonlinear Biot coefficient
%
\begin{empheq}[box=\fbox]{equation}
	B_s
	\equiv
	1
	-
	\frac{a_s}{v_{s0}}
	\left.
	\p{\bar v_s}{J}
	\right\vert_{\bar p_s,\theta,\eta_s^\beta},
	\qquad
	s\in\mc S .
	\label{eq:MC_phase_biot_coefficient}
\end{empheq}
%
The coefficient \(B_s\) is a skeleton-volume tangent evaluated in the current reacted
state.  The factor \(a_s\) maps the common skeleton Jacobian to the true solid
Jacobian and therefore carries the current pore-space allocation, packing, and
stress-free volume accommodation into the coefficient.  When the distension
split is suppressed, this reduces to the nonlinear Biot coefficient form
derived in \cite{foster2025}.  The phase Piola split is
%
\begin{equation}
	\mbf P_s'
	=
	\mbf P_s^{\prime\prime}
	+
	\left(1-B_s\right)
	\bar p_s
	\bar J_s
	(\bar{\mbf F}_s^e)^{-T}.
	\label{eq:MC_phase_pk_biot_split}
\end{equation}
%
Pushing the Piola stresses forward by
\(\bar J_s^{-1}(\cdot)(\bar{\mbf F}_s^e)^T\), and then through the same
distension map used in \eqref{eq:MC_solid_stress_definition}, leaves the
pressure correction isotropic.  Hence the phase Cauchy stress split is
%
\begin{empheq}[box=\fbox]{equation}
	\bs\sigma_s'
	=
	\bs\sigma_s^{\prime\prime}
	+
	\left(1-B_s\right)\bar p_s\mbf I,
	\qquad
	s\in\mc S .
	\label{eq:MC_phase_biot_split}
\end{empheq}
%
This phase relation does not require the solid Helmholtz energy to be written as
a sum of component energies.  The additive model in
\sref{MC_additive_component_specialization} only adds a way to compute \(B_s\)
and \(\bs\sigma_s^{\prime\prime}\) from component quantities.

\subsection{Phase and overall momentum after the Biot transform}
\label{sec:MC_phase_overall_momentum_biot}

Using the equivalent pressure \(\bar p_E=-\lambda\), the solid
volume-fraction restriction \eqref{eq:MC_solid_volume_fraction_restriction}
gives
%
\begin{equation}
	\bar p_E
	=
	-\lambda
	=
	\bar p_s,
	\qquad
	s\in\mc S .
	\label{eq:MC_phase_equivalent_pressure}
\end{equation}
%
Substituting \eqref{eq:MC_phase_biot_split} into the solid phase momentum
balance \eqref{eq:el_momentum_uniform} gives the fixed-pressure form
%
\begin{equation}
	\begin{aligned}
		\rho_s\mbf a_s
		={} &
		\nabla_{\mbf x}\cdot
		\left[
			\bs\sigma_s^{\prime\prime}
			+
			\left(1-B_s\right)\bar p_E\mbf I
			\right]
		-\phi_s\nabla_{\mbf x}\bar p_E
		+\rho_s\mbf g+\mbf b_s
		\\
		    & +
		\sum_\alpha\dot m_s^\alpha
		\left(
		\nabla_{\mbf x}\mu_s^\alpha-\mbf v_s
		\right),
		\qquad
		s\in\mc S .
	\end{aligned}
	\label{eq:MC_solid_phase_momentum_biot}
\end{equation}
%
The plasticity restrictions are written in terms of the combined reversible
stress because \(\bs\sigma_s^{\prime+}\) is the stress appearing in the residual
inequality when electrical effects are retained.  If a plastic model is
calibrated in terms of \(\bs\sigma_s^{\prime\prime}\), the phase split
\eqref{eq:MC_phase_biot_split} and the Maxwell contribution must be applied
before evaluating the plastic driving stress.

After the solid partial stresses are summed, the solid-phase Biot terms combine
as
%
\begin{align}
	\sum_{s\in\mc S}\bs\sigma_s'
	 & =
	\sum_{s\in\mc S}\bs\sigma_s^{\prime\prime}
	+
	\sum_{s\in\mc S}
	\left(1-B_s\right)\bar p_s\mbf I,
	\label{eq:MC_solid_biot_rollup_expanded}   \\
	\sum_{s\in\mc S}\bs\sigma_s'
	 & \equiv
	\bs\sigma^{\prime\prime}
	+
	\left(1-B\right)\bar p_E\mbf I,
	\label{eq:MC_solid_biot_rollup_definition} \\
	\Aboxed{
		B
	 & =
		1-\sum_{s\in\mc S}\left(1-B_s\right)
	}
	.
	\label{eq:MC_solid_biot_rollup_coefficient}
\end{align}
%
Thus the overall Biot coefficient is found after the pressure terms from the
solid phases have been added.  No extra volume-fraction weight appears in
\eqref{eq:MC_solid_biot_rollup_coefficient}, because \(B_s\) was already defined
from the phase partial Piola stress per unit skeleton reference volume.  In the
single-solid, single-fluid limit, \(B=B_s\), recovering the standard Biot form.

After summing the solid phase balances, the Biot pressure part in
\eqref{eq:MC_solid_phase_momentum_biot} can be rewritten using
\eqref{eq:MC_solid_biot_rollup_definition},
%
\begin{align*}
	 & \nabla_{\mbf x}\cdot
	\left[
		\sum_{s\in\mc S}
		\left(1-B_s\right)\bar p_E\mbf I
		\right]
	-
	\sum_{s\in\mc S}\phi_s\nabla_{\mbf x}\bar p_E
	\\
	 & \qquad =
	\nabla_{\mbf x}\cdot
	\left[
		-B\bar p_E\mbf I
		\right]
	+
	\left(
	1-\sum_{s\in\mc S}\phi_s
	\right)
	\nabla_{\mbf x}\bar p_E .
\end{align*}
%
The volume-fraction constraint identifies the coefficient of
\(\nabla_{\mbf x}\bar p_E\) in this solid remainder with the total fluid volume
fraction \(\phi=\sum_{f\in\mc F}\phi_f\).

The fluid pressure-gradient terms combine with the same equivalent pressure.
Writing \(\phi_f=\phi S_f\), using \(\sum_{f\in\mc F}S_f=1\), and using the
capillary redistribution relation implied by the interfacial energy gives
%
\begin{equation}
	-\sum_{f\in\mc F}\phi_f\nabla_{\mbf x}\bar p_f
	=
	-\phi\nabla_{\mbf x}\bar p_E .
	\label{eq:MC_fluid_pressure_gradient_rollup}
\end{equation}
%
Adding this solid remainder and the fluid contribution in
\eqref{eq:MC_fluid_pressure_gradient_rollup} gives
%
\begin{equation}
	\phi\nabla_{\mbf x}\bar p_E
	-\sum_{f\in\mc F}\phi_f\nabla_{\mbf x}\bar p_f
	=
	0 .
	\label{eq:MC_pressure_gradient_overall_rollup}
\end{equation}
%
Summing the phase momentum balances, using the pairwise internal momentum
antisymmetry \(\mbf b_{\zeta\xi}=-\mbf b_{\xi\zeta}\) in
\eqref{eq:MC_pairwise_interaction_force}, so that
\(\sum_\xi\sum_{\zeta\ne\xi}\mbf b_{\zeta\xi}=\mbf 0\), and substituting
\eqref{eq:MC_solid_biot_rollup_definition} gives the compact overall balance
%
\begin{empheq}[box=\fbox]{equation}
	\sum_\xi\rho_\xi\mbf a_\xi
	=
	\nabla_{\mbf x}\cdot
	\left[
		\bs\sigma^{\prime\prime}
		-
		B\bar p_E\mbf I
		\right]
	+\rho\mbf g
	+\sum_\xi\sum_\alpha\dot m_\xi^\alpha
	\left(
	\nabla_{\mbf x}\mu_\xi^\alpha-\mbf v_\xi
	\right).
	\label{eq:MC_overall_momentum_nonlinear_biot}
\end{empheq}

\subsection{Additive solid component specialization}
\label{sec:MC_additive_component_specialization}

The phase-level derivation does not assume that the solid Helmholtz energy is
additive.  The additive model is useful when a solid phase has one shared
skeleton motion but its stored energy can still be split into component
contributions.  Examples include adsorbed species in a
mineral framework, bound water, substitutions or defects in a mineral lattice,
and guest occupancy in hydrate cages.  In these cases the components do not
have separate solid velocities or separate deformation gradients; they share
the same \(\bar{\mbf F}_s^e\) and \(\mbf A_s^e\).  The additive split is
therefore an assumption about the energy, not a separation of the solid motion.
It also need not be an ideal-mixture assumption, because each component
potential may depend on the full composition \(\eta_s^\beta\).
Nonideal component interactions can therefore enter through the composition
dependence of \(\psi_s^\alpha\), while the generic nonadditive formulation
\eqref{eq:MC_solid_general_free_energy} remains available when such an additive
representation is too restrictive.

The additive model is obtained by choosing component potentials
\(\psi_s^\alpha\), component intrinsic densities \(\bar\rho_s^\alpha\), and
component pressures \(\bar p_s^\alpha\), then writing
%
\begin{equation}
	\psi_s
	=
	\sum_\alpha
	\eta_s^\alpha
	\psi_s^\alpha(
	\bar{\mbf F}_s^e,\mbf A_s^e,
	\bar\rho_s^\alpha,\eta_s^\beta,\theta).
	\label{eq:MC_additive_free_energy}
\end{equation}
%
Then the pressure, stress, and distension restrictions become
%
\begin{subequations}
	\label{eq:MC_additive_mixture_rules}
	\begin{align}
		\bar p_s
		 & =
		\sum_\alpha
		\phi_s^\alpha\bar p_s^\alpha,
		\label{eq:MC_additive_pressure_average}   \\
		\bar p_s^\alpha
		 & =
		(\bar\rho_s^\alpha)^2
		\left.
		\p{\psi_s^\alpha}{\bar\rho_s^\alpha}
		\right\vert_{\mc X_s},
		\label{eq:MC_additive_component_pressure} \\
		\bs\sigma_s'
		 & =
		\sum_\alpha
		\bs\sigma_s^{\prime\alpha},
		\label{eq:MC_additive_stress_sum}         \\
		\bs\sigma_s^{\prime\alpha}
		 & =
		\rho_s^\alpha
		\mbf A_s^{-T}
		\left(
		\left.
		\p{\psi_s^\alpha}{\bar{\mbf F}_s^e}
		\right\vert_{\mc X_s}
		(\bar{\mbf F}_s^e)^T
		\right)
		\mbf A_s^T,
		\label{eq:MC_additive_component_stress}   \\
		\bs\sigma_s'
		 & =
		\rho_s
		\sum_\alpha
		\eta_s^\alpha
		\left.
		\p{\psi_s^\alpha}{\mbf A_s^e}
		\right\vert_{\mc X_s}
		(\mbf A_s^e)^T .
		\label{eq:MC_additive_distension_force}
	\end{align}
\end{subequations}
%
The component pressures in
\eqref{eq:MC_additive_component_pressure} can also be read from the storage
multipliers.  The phase-level restriction
\eqref{eq:MC_phase_equivalent_pressure} fixes only the sum
\(\sum_\alpha\pi_s^\alpha/\phi_s=\bar p_s\).  In the additive model, each term
in that sum is matched to the component's share of the phase-pressure average,
%
\begin{equation*}
	\frac{\pi_s^\alpha}{\phi_s}
	=
	\phi_s^\alpha\bar p_s^\alpha,
	\qquad
	\bar p_s^\alpha
	=
	\frac{\pi_s^\alpha}{\phi_s\phi_s^\alpha},
	\qquad
	s\in\mc S .
\end{equation*}
%
Summing this identity over components recovers
\eqref{eq:MC_additive_pressure_average} and
\eqref{eq:MC_phase_equivalent_pressure}.  Thus the component pressures are
determined once the storage multipliers and component solid-volume
fractions are known.

The residual inequality uses the phase plastic powers.  A conservative
component-resolved plastic law therefore lets each component change the mobility
of the common phase plastic rates, rather than giving each component its own
plastic motion.  One sufficient additive form is
%
\begin{subequations}
	\label{eq:MC_component_resolved_plastic_flow_rules}
	\begin{align}
		\dot{\bar{\mbf F}}_s^p(\bar{\mbf F}_s^p)^{-1}
		 & =
		\left(
		\sum_\alpha
		\Lambda_{\bar{\mbf F}_s}^\alpha(\mc X_s)
		\right)
		(\mbf S_s^{+})^T,
		\label{eq:MC_component_true_plastic_flow_rule}       \\
		\dot{\mbf A}_s^p(\mbf A_s^p)^{-1}
		 & =
		\left(
		\sum_\alpha
		\Lambda_{\mbf A_s}^\alpha(\mc X_s)
		\right)
		(\mbf A_s^e)^T
		\bs\sigma_s^{\prime+}
		(\mbf A_s^e)^{-T},
		\label{eq:MC_component_distension_plastic_flow_rule} \\
		\sum_\alpha
		\Lambda_{\bar{\mbf F}_s}^\alpha(\mc X_s)
		 & \ge0,
		\qquad
		\sum_\alpha
		\Lambda_{\mbf A_s}^\alpha(\mc X_s)
		\ge0 .
		\label{eq:MC_component_plastic_mobilities}
	\end{align}
\end{subequations}
%
These rules do not assign separate solid velocities or separate total
deformation gradients to the components.  They assign separate component
mobility contributions to the common solid plastic rates.  Only the summed
mobilities enter the phase plastic powers, so the sums must be nonnegative;
requiring each component mobility to be nonnegative is the stronger
specialization in which each component contribution is separately dissipative.
More general component-indexed flow directions are admissible only if the
resulting combined plastic power in \eqref{eq:MC_residual_dissipation} remains
nonnegative.

\subsubsection{Component reaction-modified Biot coefficient derivation}
\label{sec:MC_component_biot_derivation}

The additive rules \eqref{eq:MC_additive_mixture_rules} give the component
single-prime stress.  The pressure-based momentum balance needs the
double-prime stress, so the fixed-pressure Legendre transform is applied to each
component before the component stresses are summed.
%
Let \(\bar v_s^\alpha=1/\bar\rho_s^\alpha\),
\(v_{s0}^\alpha=1/\rho_{s0}^\alpha\), and
\(\widetilde\psi_s^\alpha
=\psi_s^\alpha+\bar p_s^\alpha\bar v_s^\alpha\).  The component
fixed-pressure Legendre transform follows the same steps as the phase-level
transform in \S\ref{sec:MC_phase_biot_legendre_transform}, with
\(\bar p_s^\alpha\) replacing \(\bar p_s\) and \(\bar v_s^\alpha\) replacing
\(\bar v_s\).  The component reference energy per unit skeleton reference
volume is \(\rho_{s0}^\alpha\psi_s^\alpha\), so the single-prime and
double-prime component partial Piola stresses are
%
\begin{subequations}
	\label{eq:MC_additive_component_pk_stresses}
	\begin{align}
		\mbf P_s^{\prime\alpha}
		 & =
		\rho_{s0}^\alpha
		\left.
		\p{\widetilde\psi_s^\alpha}{\bar{\mbf F}_s^e}
		\right\vert_{\bar p_s^\alpha,\theta,\eta_s^\beta}
		\label{eq:MC_additive_component_single_prime_pk_stress} \\
		\rho_{s0}^\alpha
		\left.
		\p{\widetilde\psi_s^\alpha}{\bar{\mbf F}_s^e}
		\right\vert_{\bar p_s^\alpha,\theta,\eta_s^\beta}
		 & =
		\rho_{s0}^\alpha
		\left.
		\p{\psi_s^\alpha}{\bar{\mbf F}_s^e}
		\right\vert_{\bar v_s^\alpha,\eta_s^\beta,\theta},
		\label{eq:MC_additive_component_legendre_pk_identity}   \\
		\mbf P_s^{\prime\prime\alpha}
		 & =
		\rho_{s0}^\alpha
		\left.
		\p{\psi_s^\alpha}{\bar{\mbf F}_s^e}
		\right\vert_{\bar p_s^\alpha,\theta,\eta_s^\beta}.
		\label{eq:MC_additive_component_double_prime_pk_stress}
	\end{align}
\end{subequations}
%
Using the Legendre transform at fixed \(\bar p_s^\alpha\),
%
\begin{equation}
	\mbf P_s^{\prime\alpha}
	=
	\mbf P_s^{\prime\prime\alpha}
	+
	\bar p_s^\alpha\rho_{s0}^\alpha
	\left.
	\p{\bar v_s^\alpha}{\bar{\mbf F}_s^e}
	\right\vert_{\bar p_s^\alpha,\theta,\eta_s^\beta}.
	\label{eq:MC_additive_component_pk_legendre_split}
\end{equation}
%
Using the same \(J=a_s\bar J_s\) chain rule as in the phase-level derivation,
the pressure correction in \eqref{eq:MC_additive_component_pk_legendre_split}
identifies the component nonlinear Biot coefficient
%
\begin{empheq}[box=\fbox]{equation}
	B_s^\alpha
	\equiv
	1
	-
	\frac{a_s}{v_{s0}^\alpha}
	\left.
	\p{\bar v_s^\alpha}{J}
	\right\vert_{\bar p_s^\alpha,\theta,\eta_s^\beta}.
	\label{eq:MC_additive_component_biot}
\end{empheq}
%
where \(B_s^\alpha\) is the component nonlinear Biot coefficient.  This gives the
component Piola split
%
\begin{equation}
	\mbf P_s^{\prime\alpha}
	=
	\mbf P_s^{\prime\prime\alpha}
	+
	(1-B_s^\alpha)\bar p_s^\alpha
	\bar J_s
	(\bar{\mbf F}_s^e)^{-T}.
	\label{eq:MC_additive_component_pk_biot_split}
\end{equation}
%
Pushing either component Piola stress forward by the operation
\(\bar J_s^{-1}(\cdot)(\bar{\mbf F}_s^e)^T\), and then through the same
distension map used in \eqref{eq:MC_additive_component_stress}, leaves the
pressure correction isotropic.  Hence the component Cauchy stress split is
%
\begin{equation}
	\bs\sigma_s^{\prime\alpha}
	=
	\bs\sigma_s^{\prime\prime\alpha}
	+
	(1-B_s^\alpha)\bar p_s^\alpha\mbf I .
	\label{eq:MC_additive_component_biot_split}
\end{equation}
%
Summing over components gives the phase stress split
%
\begin{subequations}
	\label{eq:MC_additive_biot_mixture_rule}
	\begin{align}
		\bs\sigma_s^{\prime\prime}
		 & =
		\sum_\alpha
		\bs\sigma_s^{\prime\prime\alpha},
		\label{eq:MC_additive_double_prime_sum} \\
		\bs\sigma_s'
		 & =
		\bs\sigma_s^{\prime\prime}
		+
		(1-B_s)\bar p_s\mbf I,
		\label{eq:MC_additive_phase_biot_split}
	\end{align}
	\begin{empheq}[box=\fbox]{equation}
		B_s
		=
		1
		-
		\frac{1}{\bar p_s}
		\sum_\alpha
		(1-B_s^\alpha)\bar p_s^\alpha.
		\label{eq:MC_additive_biot_mixture_coefficient}
	\end{empheq}
\end{subequations}
%
These additive rules give component formulas for the same phase stress split.
They keep the plastic, distension, thermal, and composition-dependent
stress-free terms from the full entropy inequality.

\subsection{Single-component solid specialization}
\label{sec:MC_single_component_rollup}

The single-component mineral solid is recovered from the additive theory by
choosing one active component in each solid phase and imposing
%
\begin{equation}
	\eta_s^s=1,
	\qquad
	\dot\eta_s^s=0,
	\qquad
	\phi_s^s=1,
	\qquad
	\bar\rho_s^s=\bar\rho_s .
	\label{eq:MC_single_component_constraints}
\end{equation}
%
Then \eqref{eq:MC_solid_general_free_energy} reduces to the one-component solid
free energy
%
\begin{equation}
	\psi_s
	=
	\psi_s(
	\bar{\mbf F}_s^e,\mbf A_s^e,\bar\rho_s,\theta),
	\label{eq:MC_single_component_free_energy}
\end{equation}
%
the solid composition projection \eqref{eq:MC_solid_composition_projection}
vanishes identically, and the additive component rules reduce to
%
\begin{equation}
	\bar p_s^s=\bar p_s,
	\qquad
	\bs\sigma_s^{\prime s}=\bs\sigma_s',
	\qquad
	\bs\sigma_s^{\prime\prime s}=\bs\sigma_s^{\prime\prime},
	\qquad
	B_s^s=B_s .
	\label{eq:MC_single_component_reductions}
\end{equation}
%
This shows that the one-component solid is not a separate theory.  It is the
one-component limit of the multicomponent solid.  The stress-free thermal maps,
plastic variables, distension variables, and source terms still belong to the
phase, but there is no independent solid-composition variation.

For this specialization, the phase-level Biot coefficient
\eqref{eq:MC_phase_biot_coefficient}, the phase stress split
\eqref{eq:MC_phase_biot_split}, the solid phase momentum balance
\eqref{eq:MC_solid_phase_momentum_biot}, and the overall balance
\eqref{eq:MC_overall_momentum_nonlinear_biot} reduce to the one-component
solid formulas.  In the single-solid, single-fluid limit, the overall mixture
coefficient
\eqref{eq:MC_solid_biot_rollup_coefficient} gives \(B=B_s\).

\subsection{Scalar distension simplifications}
\label{sec:MC_scalar_distension_simplifications}

The tensor distension theory reduces to a scalar factor when the pore-structure
change is isotropic.  Scalar distension changes local solid volume but not the
shape of the pore structure.  Anisotropic pore collapse or fabric evolution
requires the tensor \(\mbf A_s\), or another internal variable.  The scalar
specialization is imposed on the elastic, plastic, and stress-free distension
factors,
%
\begin{equation}
	\mbf A_s^\bullet=(a_s^\bullet)^{1/3}\mbf I,
	\qquad
	\bullet\in\{e,p,0\},
	\qquad
	s\in\mc S .
	\label{eq:MC_scalar_distension_factors}
\end{equation}
%
The factors multiply,
%
\begin{equation}
	a_s=a_s^e a_s^p a_s^0,
	\qquad
	s\in\mc S ,
	\label{eq:MC_scalar_distension_product}
\end{equation}
%
and the true solid deformation relation becomes
%
\begin{equation}
	\mbf F=a_s^{1/3}\bar{\mbf F}_s,
	\qquad
	J=a_s\bar J_s .
	\label{eq:MC_scalar_distension_deformation}
\end{equation}
%
For a generic multicomponent solid, the tensor distension restriction
\eqref{eq:MC_distension_force_definition} reduces to the trace condition
%
\begin{equation}
		\frac{1}{3}\operatorname{tr}\left(\bs\sigma_s^{\prime+}\right)
	=
	\rho_s
	\left.
	\p{\psi_s}{a_s^e}
	\right\vert_{\mc X_s}
	a_s^e,
	\qquad
	s\in\mc S .
	\label{eq:MC_scalar_distension_stress_restriction}
\end{equation}
%
Using the phase-level Biot transform, the same condition can be written as a
restriction on the phase double-prime trace,
%
\begin{equation}
	\frac{1}{3}\operatorname{tr}(\bs\sigma_s^{\prime\prime})
	=
	\rho_s
	\left.
	\p{\psi_s}{a_s^e}
	\right\vert_{\mc X_s}
	a_s^e
	-\left(1-B_s\right)\bar p_s,
	\qquad
	s\in\mc S .
	\label{eq:MC_scalar_double_prime_distension_trace}
\end{equation}
%
These are not additional evolution equations for \(a_s\).  They are the scalar
form of the reversible tensor admissibility condition.  The scalar specialization
also reduces the plastic-distension power in
\eqref{eq:MC_residual_dissipation} to
%
\begin{equation}
		\sum_{s\in\mc S}\bs\sigma_s^{\prime+}:
	\mbf A_s^e\dot{\mbf A}_s^p
	(\mbf A_s^p)^{-1}(\mbf A_s^e)^{-1}
	=
	\sum_{s\in\mc S}
		\frac{1}{3}\operatorname{tr}\left(\bs\sigma_s^{\prime+}\right)
	\frac{\dot a_s^p}{a_s^p}.
	\label{eq:MC_scalar_plastic_distension_power}
\end{equation}
%
Thus scalar plastic distension is hydrostatic compaction or dilation driven by
the single-prime mean stress.

\subsection{Distension and crystallization pressure}
\label{sec:MC_distension_crystallization_pressure}

Crystallization pressure is not an extra force added to
\eqref{eq:MC_overall_momentum_nonlinear_biot}.  It must be represented either by
an admissible solid free energy, by an admissible kinetic law, or by an
algebraic affinity restriction.  In the scalar setting, the volumetric split is
%
\begin{equation}
	\ln J=\ln a_s+\ln\bar J_s,
	\qquad
	s\in\mc S .
	\label{eq:MC_crystallization_volumetric_split}
\end{equation}
%
Therefore a volumetric penalty written only in terms of \(\ln a_s\), or only in
terms of \(\ln\bar J_s\), does not generally reproduce the scalar
Coleman--Noll trace condition.  The transformed fixed-pressure volumetric
potential should generate the double-prime stress; the Biot pressure correction
has already been accounted for in \eqref{eq:MC_phase_biot_split}.  A
simple scalar choice, with tangent bulk modulus \(K_s^\star\), is
%
\begin{equation}
	\widetilde W_{{\rm vol},s}
	=
	\frac{K_s^\star}{2}
	\left(\ln J\right)^2 .
	\label{eq:MC_component_transformed_volumetric_energy}
\end{equation}
%
Holding \(\bar J_s\) and \(\bar p_s\) fixed,
%
\begin{equation}
	a_s
	\left.
	\p{\widetilde W_{{\rm vol},s}}{a_s}
	\right\vert_{\bar J_s,\bar p_s}
	=
	K_s^\star\ln J .
	\label{eq:MC_component_transformed_volumetric_trace}
\end{equation}
%
Thus the transformed volumetric potential supplies
%
\begin{equation}
	\frac{1}{3}\operatorname{tr}(\bs\sigma_s^{\prime\prime})
	=
	K_s^\star\ln J ,
	\label{eq:MC_component_double_prime_trace_from_energy}
\end{equation}
%
and the original Helmholtz trace required by the scalar distension restriction is
%
\begin{equation}
	\rho_s
	\left.
	\p{\psi_s}{a_s^e}
	\right\vert_{\mc X_s}
	a_s^e
	=
	K_s^\star\ln J
	+\left(1-B_s\right)\bar p_s .
	\label{eq:MC_component_cn_trace_satisfied}
\end{equation}
%
This is exactly \eqref{eq:MC_scalar_double_prime_distension_trace}.  Once the
stress generated by the energy is inserted into the momentum equation, no
separate crystallization-pressure load should be inserted.

Crystallization can instead be prescribed or diagnosed through the mechanism
affinity.  For a mechanism \((m)\) precipitating solid phase \(s_{(m)}\),
%
\begin{equation}
	\mc A_{(m)}
	=
	-
	\sum_\xi\sum_\alpha
	\mu_\xi^\alpha\nu_{\xi (m)}^\alpha,
	\qquad
	\mc A_{(m)}\dot r_{(m)}\ge0 .
	\label{eq:MC_crystallization_affinity_kinetics}
\end{equation}
%
If a classical crystallization pressure \(\bar p_{\text{xtal},s_{(m)}}\) is used, it is
best interpreted as an algebraic affinity law,
%
\begin{equation}
	\mc A_{(m)}
	=
	\bar v_{s_{(m)}}
	\left[
	\bar p_{\text{xtal},s_{(m)}}
	-\frac{1}{3}
	\operatorname{tr}
	\left(
	\bs\sigma_{s_{(m)}}'
	\right)
	\right],
	\qquad
	\bar v_{s_{(m)}}=\frac{1}{\bar\rho_{s_{(m)}}}.
	\label{eq:MC_crystallization_affinity_volumetric}
\end{equation}
%
For example, the classical supersaturation formula
\(\bar p_{\text{xtal},s_{(m)}}=R\theta\ln\Omega_{(m)}/V_{s_{(m)}}\), where
\(R\) is the gas constant, \(\Omega_{(m)}\) is the supersaturation ratio, and
\(V_{s_{(m)}}\) is the molar volume of the precipitating solid, gives
%
\begin{equation}
	\mc A_{(m)}
	=
	\bar v_{s_{(m)}}
	\left[
		\frac{R\theta}{V_{s_{(m)}}}
		\ln\Omega_{(m)}
		-\frac{1}{3}
		\operatorname{tr}
		\left(
		\bs\sigma_{s_{(m)}}'
		\right)
		\right].
	\label{eq:MC_crystallization_supersaturation_affinity}
\end{equation}
%
At reversible equilibrium, \(\mc A_{(m)}=0\), and the classical
crystallization pressure equals the admissible current phase trace stress.  If
the affinity and stress are solved instead, \(\Omega_{(m)}\) is a diagnostic
output.
