\section{Solid-reference form for reservoir simulation}
\label{sec:reservoir_simulation}

This section rewrites the transport and mechanics equations in a form suitable
for solution on the trajectory of the solid skeleton.  The solid motion
$\bs\chi$ supplies the computational map from the solid reference coordinate
$\mbf X$ to the current coordinate $\mbf x$, with deformation gradient
$\mbf F$ and Jacobian $J$.

\subsection{Relative flux vectors}
\label{sec:relative_flux_vectors}

The phase momentum equation \eqref{eq:el_mom_f_simplified} determines
Darcy-scale relative fluxes once interphase drag is written in terms of a
velocity relative to the solid skeleton.  For each fluid phase \(f\), define the
current relative mass flux by
%
\begin{equation}
	\mbf w_f
	=\phi_f\bar\rho_f\left(\mbf v_f-\mbf v_s\right)
	=\rho_f\left(\mbf v_f-\mbf v_s\right).
	\label{eq:relative_mass_flux_definition}
\end{equation}
%
Thus
%
\begin{equation}
	\mbf v_f
	=\mbf v_s+\frac{\mbf w_f}{\rho_f}.
	\label{eq:relative_flux_velocity_reconstruction}
\end{equation}
%
A linear interphase-drag closure represents the part of the fluid-phase
interaction force produced by motion through the skeleton as
%
\begin{equation}
	\mbf b_f
	=-\frac{\mu_f\phi_f}{\bar\rho_f}\,
	\bs\kappa_f^{-1}\mbf w_f ,
	\label{eq:fluid_phase_drag_law}
\end{equation}
%
where $\mu_f$ is the dynamic viscosity and $\bs\kappa_f$ is the permeability
tensor for phase $f$.  Phase relative permeability may be absorbed into
$\bs\kappa_f$.

Substituting \eqref{eq:relative_flux_velocity_reconstruction} and
\eqref{eq:fluid_phase_drag_law} into the fluid momentum balance
\eqref{eq:el_mom_f_simplified} gives, for each fluid phase,
%
\begin{align}
	\frac{1}{\rho_f}
	\left(
		\phi_f^2\mu_f\bs\kappa_f^{-1}
		+\sum_\alpha\dot c_f^\alpha\mbf I
	\right)\mbf w_f
	= & \;
	\rho_f\left(\mbf g-\mbf a_f\right)
	-\phi_f\nabla_{\mbf x}\bar p_f
	\notag                          \\
	  & +\sum_\alpha\dot c_f^\alpha
	\left(\nabla_{\mbf x}\tau-\mbf v_s\right),
	\label{eq:relative_flux_linear_system}
\end{align}
%
The isotropic rate correction contains the conversion insertion rate.  Transport
accumulation is dissipative and is closed through
\eqref{eq:MC_relative_transport_closures}; only conversion insertion contributes
to the momentum term involving \(\nabla_{\mbf x}\tau\).

Define the modified permeability tensor
%
\begin{equation}
	\widetilde{\bs\kappa}_f
	=
	\left(
		\phi_f^2\mu_f\bs\kappa_f^{-1}
		+\sum_\alpha\dot c_f^\alpha\mbf I
	\right)^{-1}.
	\label{eq:modified_relative_flux_permeability}
\end{equation}
%
Then the admissible relative mass flux is
%
\begin{empheq}[box=\fbox]{align}
	\mbf w_f
	= \rho_f\widetilde{\bs\kappa}_f
	\left(
	\rho_f\left(\mbf g-\mbf a_f\right)
		-\phi_f\nabla_{\mbf x}\bar p_f
		+\sum_\alpha\dot c_f^\alpha
		\left(\nabla_{\mbf x}\tau-\mbf v_s\right)
	\right).
	\label{eq:generalized_fluid_darcy_mass_flux}
\end{empheq}
%
This expression is the multicomponent fluid-phase analogue of a generalized
Darcy flux.  The first two terms are the body-force and pressure-gradient
contributions, while the last term is the momentum contribution from conversion
insertion.  Chemical-potential gradients drive component-relative transport
through \eqref{eq:MC_relative_transport_closures}; that dissipative role is
separate from the role of \(\tau\) in the conversion variation.

When conversion-insertion effects are negligible,
\(\dot c_f^\alpha=0\), and when fluid-phase inertia is negligible,
\eqref{eq:generalized_fluid_darcy_mass_flux} reduces to
%
\begin{equation}
	\phi_f\left(\mbf v_f-\mbf v_s\right)
	=-\frac{\bs\kappa_f}{\mu_f}
	\left(\nabla_{\mbf x}\bar p_f-\bar\rho_f\mbf g\right).
	\label{eq:standard_darcy_mass_flux_limit}
\end{equation}
%
This is Darcy's law for phase \(f\) relative to the deforming solid skeleton.
The more general law \eqref{eq:generalized_fluid_darcy_mass_flux} applies when
net component accumulation rates are large enough for momentum production to
affect phase transport.

\subsection{Solid-reference relative fluxes}
\label{sec:solid_reference_relative_fluxes}

The current relative flux in \eqref{eq:generalized_fluid_darcy_mass_flux} is
converted to a flux through surfaces in the solid reference configuration by the
Piola transform.  Define the reference relative mass flux
%
\begin{equation}
	\mbf W_f
	=
	J\mbf F^{-1}\mbf w_f .
	\label{eq:reference_relative_mass_flux}
\end{equation}
%
Define the pulled-back modified permeability tensor
%
\begin{equation}
	\widetilde{\mbf K}_f
	=
	J\mbf F^{-1}
	\widetilde{\bs\kappa}_f
	\mbf F^{-T}.
	\label{eq:pulled_back_modified_permeability}
\end{equation}
%
The flux assembled in the reference component balances is then
%
\begin{align}
	\mbf W_f
	= & \; \rho_f\widetilde{\mbf K}_f
	\left[
		\rho_f\mbf F^T\left(\mbf g-\mbf a_f\right)
		-\phi_f\nabla_{\mbf X}\bar p_f
		+\sum_\alpha\dot c_f^\alpha
		\left(\nabla_{\mbf X}\tau-\mbf F^T\mbf v_s\right)
		\right].
	\label{eq:solid_reference_relative_flux}
\end{align}
%
Here gradients of scalar fields obey
$\nabla_{\mbf x}(\cdot)=\mbf F^{-T}\nabla_{\mbf X}(\cdot)$, while the
divergence of a spatial flux obeys
%
\begin{equation}
	J\nabla_{\mbf x}\cdot\mbf w_f
	=
	\nabla_{\mbf X}\cdot
	\mbf W_f .
	\label{eq:solid_reference_flux_divergence}
\end{equation}
%
Equations~\eqref{eq:reference_relative_mass_flux}--\eqref{eq:solid_reference_relative_flux}
and \eqref{eq:solid_reference_flux_divergence} give the reference flux used in
the multicomponent balances.  The expression retains phase labels, component
conversion rates, and gradients of \(\tau\).

\subsection{Component balances on the solid trajectory}
\label{sec:solid_reference_component_balances}

For a fluid phase, substituting
\eqref{eq:relative_flux_velocity_reconstruction} into
\eqref{eq:averaged_component_spatial_mass}, using
\eqref{eq:component_partial_density_representation}, and writing the storage
term on the solid trajectory gives
%
\begin{equation}
	\left.
	\frac{\partial}{\partial t}
	\left(J\phi_f\bar\rho_f\eta_f^\alpha\right)
	\right\vert_{\mbf X}
	+\nabla_{\mbf X}\cdot
	\left[
	\eta_f^\alpha\mbf W_f
	+
	J\mbf F^{-1}
	\left(
	\mbf j_{f,{\rm disp}}^\alpha
	+
	\mbf j_{f,{\rm diff}}^\alpha
	\right)
	\right]
	=
	J\sum_m\nu_{f (m)}^\alpha\dot r_{(m)} .
	\label{eq:solid_reference_fluid_component_balance}
\end{equation}
%
For each one-component solid phase \(s\), the phase velocity is the skeleton velocity,
so the relative-flux term is absent,
%
\begin{equation}
	\left.
	\frac{\partial}{\partial t}
	\left(J\phi_s\bar\rho_s^s\right)
	\right\vert_{\mbf X}
	=
	J\sum_m\nu_{s (m)}^s\dot r_{(m)},
	\qquad
	s\in\mc S .
	\label{eq:solid_reference_solid_component_balance}
\end{equation}
%
Thus the solid-reference storage variables are the Lagrangian fluid component
masses \(J\phi_f\bar\rho_f\eta_f^\alpha\) and the Lagrangian solid phase masses
\(J\phi_s\bar\rho_s^s\).  Transport through the solid mesh contains both bulk
fluid motion relative to the skeleton and component diffusion or dispersion
relative to the fluid phase.

\subsection{Overall momentum on the solid trajectory}
\label{sec:solid_reference_overall_momentum}

The overall momentum balance \eqref{eq:MC_overall_momentum_nonlinear_biot} can be
written on the solid trajectory by applying the Piola transform to the stress
divergence and multiplying the remaining spatial source terms by $J$.  Let
\(\mbf P^{\prime\prime}=J\bs\sigma^{\prime\prime}\mbf F^{-T}\) denote the
first Piola--Kirchhoff stress associated with the fixed-pressure skeleton
stress.  This transformation gives
%
\begin{align}
	J\sum_\xi\rho_\xi\mbf a_\xi
	= & \;
	\nabla_{\mbf X}\cdot
	\left(
	\mbf{P}^{\prime\prime}-B\bar p_E
	J \mbf F^{-T}
	\right)
	+J\rho\mbf g
	+J\sum_\xi\sum_\alpha\dot c_\xi^\alpha
	\left(\nabla_{\mbf x}\tau-\mbf v_\xi\right).
	\label{eq:solid_reference_overall_momentum}
\end{align}
%
If a quasi-static solid-skeleton solve is desired, the inertial term on the left
is dropped.  If the fluid velocities are eliminated with
\eqref{eq:relative_flux_velocity_reconstruction}, then the same equation is
solved with $\mbf v_f=\mbf v_s+\mbf w_f/\rho_f$ for fluid phases and
$\mbf v_s$ for the solid phase.  This leaves the solid displacement, the
Lagrangian partial masses, and the fluid relative fluxes as the natural fields
for a reference-configuration implementation.

\subsection{Summary of the reservoir-simulation solve block}
\label{sec:reservoir_simulation_solve_block}

For a solid-reference reservoir simulator, the coupled solve can be assembled
from the overall momentum balance, the component storage equations, the
relative flux reconstruction, and the capillary-pressure relations.  In the
general finite-deformation form, the mechanics equation is
\eqref{eq:solid_reference_overall_momentum},
%
\begin{align}
	\mbf{0}
	= & \;
	\nabla_{\mbf X}\cdot
	\left(
	\mbf{P}^{\prime\prime}-B\bar p_E
	J \mbf F^{-T}
	\right)
	+J\rho\mbf g
	+J\sum_\xi\sum_\alpha\dot c_\xi^\alpha
	\left(\nabla_{\mbf x}\tau-\mbf v_\xi\right).
	\label{eq:reservoir_simulation_overall_momentum_summary}
\end{align}
%
The fluid component balances on the same solid trajectory are
%
\begin{equation}
	\left.
	\frac{\partial}{\partial t}
	\left(J\phi_f\bar\rho_f\eta_f^\alpha\right)
	\right\vert_{\mbf X}
	+\nabla_{\mbf X}\cdot
	\left[
	\eta_f^\alpha\mbf W_f
	+
	J\mbf F^{-1}
	\left(
	\mbf j_{f,{\rm disp}}^\alpha
	+
	\mbf j_{f,{\rm diff}}^\alpha
	\right)
	\right]
	=
	J\sum_m\nu_{f (m)}^\alpha\dot r_{(m)} ,
	\qquad f\in\mc F ,
	\label{eq:reservoir_simulation_fluid_component_summary}
\end{equation}
%
with current velocities reconstructed from the relative mass fluxes by
%
\begin{equation}
	\mbf v_f
	=\mbf v_s+\frac{\mbf w_f}{\rho_f},
	\qquad
	\mbf W_f=J\mbf F^{-1}\mbf w_f .
	\label{eq:reservoir_simulation_velocity_flux_summary}
\end{equation}
%
The corresponding admissible reference flux is
%
\begin{align}
	\mbf W_f
	= & \; \rho_f\widetilde{\mbf K}_f
	\left[
		\begin{aligned}
			&\rho_f\mbf F^T\left(\mbf g-\mbf a_f\right)
			-\phi_f\nabla_{\mbf X}\bar p_f
			+\sum_\alpha\dot c_f^\alpha
			\left(\nabla_{\mbf X}\tau-\mbf F^T\mbf v_s\right)
		\end{aligned}
		\right],
	\qquad f\in\mc F .
	\label{eq:reservoir_simulation_reference_flux_summary}
\end{align}
%
When component mass-addition effects and fluid inertia are negligible, this reduces
to the familiar Darcy relative flux in
\eqref{eq:standard_darcy_mass_flux_limit}.  Capillary, or electrocapillary,
pressure closes the phase pressures algebraically through the saturation state,
%
\begin{equation}
	\phi = \sum_{f\in\mc F}\phi_f,
	\qquad
	S_f = \frac{\phi_f}{\phi},
	\qquad
	p_{c,nw}
	=
	\bar p_n-\bar p_w
	-\left(\omega_n^+-\omega_w^+\right),
	\qquad (w,n)\in\mc I .
	\label{eq:reservoir_simulation_capillary_summary}
\end{equation}
%
Equivalently, a simulator may choose one reference pressure and reconstruct the
other fluid pressures from the selected capillary-pressure laws before
evaluating the fluxes.  In a neutral model, \(\omega_f^+=0\) and the summary
reduces to the usual relation \(p_{c,nw}=\bar p_n-\bar p_w\).  This algebraic
reconstruction also makes the saturation-weighted pressure gradient collapse to
the equivalent-pressure gradient in \eqref{eq:MC_fluid_pressure_gradient_rollup}.
