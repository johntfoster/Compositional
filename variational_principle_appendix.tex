\section{Variational Derivation}
\label{app:variational}
\label{sec:variational}

We derive the governing equations from Hamilton's extended principle, following Bedford \& Drumheller~\cite{drumheller1980thermomechanical} and Drumheller~\cite{drumheller2000}, adapted to a compositional mixture system containing phases ($N$ solid, $1$ aquous, $1$ oelic, and $1$ gas phases) and arbitrary miscible components $\alpha$ within each phase.

Each phase has volume fraction~$\phi_\xi$, intrinsic density~$\bar\rho_\xi$, bulk density $\rho_\xi = \phi_\xi\bar\rho_\xi$, velocity~$\mbf{v}_\xi$, deformation gradient~$\mbf{F}_\xi$, and Jacobian $J_\xi = \det(\mbf{F}_\xi)$.  

\subsection{Variational principle and its terms}

The extended Hamilton's principle is
\begin{equation*}
  \int_{t_1}^{t_2}\left[\delta(T - U) + \delta W + \sum_k \delta C_k\right] dt = 0,
\end{equation*}
with kinetic energy, internal energy, virtual work, and four constraints:
\begin{align}
T &= \sum_\xi \int_{\mc{B}}\frac{1}{2} \rho_\xi \mbf{v}_\xi \cdot \mbf{v}_\xi \dv,
  \label{eq:T_def} \\
  U &= \sum_{\xi = \lbrace \xi_s, \xi_f \rbrace} \left[ \int_{\mc{B}_{s 0}} \rho_{\xi_{s0}}\,\psi_{\xi_{s}}(\mbf{F}_{\xi_s}, \bar\rho_{\xi_s}) \,  dV_{\xi_s}
     + \int_{\mc{B}} \rho_{\xi_f}\,\psi_{\xi_f}(\bar\rho_{\xi_f}, \eta^\al_{\xi_f})  \dv \right] \notag \\ 
    &\qquad + \int_{\mc{B}} \gamma(S_a, S_g) \dv, 
  \label{eq:U_def} \\
  \delta W &= \sum_\xi \int_{\mc{B}} \left[(\rho_\xi\mbf{g} + \mbf{b}_\xi)\cdot\delta\mbf{x}_\xi + \sum_\alpha \mc{L}^\alpha_\xi\,\frac{\delta c^\alpha_\xi}{J_\xi}\right] \,\dv,
  \label{eq:W_def} \\
  C_1 &= \int_{\mc{B}} \lambda\left(\sum_\xi\phi_\xi - 1\right) \,\dv,
  \label{eq:C1_def} \\
  C_2 &= \sum_\xi \sum_\al \int_{\mc{B}_{\xi 0}} \mu^\al_\xi \left( J_\xi - \frac{\phi_{\xi 0} \eta^\al_{\xi 0} \bar\rho_{\xi 0} + c_\xi^\alpha}{\phi_\xi \eta^\al_\xi \bar\rho_\xi} \right) \dV_\xi, 
  \label{eq:C2_def} \\
  C_3 &= \sum_\xi \int_{\mc{B}} \pi_\xi \left( \sum_\alpha \eta_\xi^\alpha - 1 \right) \dv,
  \label{eq:C3_def} \\
  C_4 &= \sum_\alpha \int_{\mc{B}} \tau^\alpha \sum_\xi  \left.\hat{c}^\al_\xi\right\vert_\alpha \dv. \\
  C_5 &= \int_{\mc{B}} \zeta \sum_\xi \sum_\alpha  \hat{c}^\al_\xi \dv.
  \label{eq:C4_def}
\end{align}
%
where $\xi_s$ is the subset of $\xi$ of single-component immicible solid phases, e.g.\ $\xi_s = \lbrace A, B \rbrace$ where $A$ is a reactant and $B$ is a solid product. $\xi_f = \lbrace a, o, g \rbrace$ are the mobile aqueous, oeliec, and gas phases.  The Helmholtz free energies are $\psi_{\xi_s}$, $\psi_{\xi_f}$.  The interaction forces satisfy $\sum_\xi \mbf{b}_\xi = \mbf{0}$, and $\mc{L}^\alpha_\xi$ is a generalized force conjugate to mass transfer (identified later via Coleman--Noll).  The Lagrange multipliers~$\lambda$, $\mu^\alpha_\xi$, $\pi_\xi$, and~$\tau^\alpha$ enforce the volume fraction constraint, component mass conservation, component mass partitioning within a phase, and component-wise mixture mass balance, respectively. Temperature is not treated as a kinematic variable in the variational principle; any thermal constitutive restrictions are deferred to the mixture entropy inequality.

We treat $\mbf{x}_\xi$, $\eta^\alpha_\xi$, $\phi_\xi$, $\bar\rho_\xi$, and $c_\xi^\alpha$ as independent fields (subject to constraints) and compute the variations in what follows.

\subsection{Extending Beford and Drumheller identity 1.28}

Below we extend the useful identity (1.28) of Bedford and Drumheller \cite{drumheller1980thermomechanical} to the present notation and for components as the conserved quantity, we write
%
\begin{align}
  \delta \int_{\mc{B}} \rho^\alpha_\xi \Gamma_\xi \, \dv
  &= \delta \int_{\mc{B}} \left(\rho^\alpha_{\xi 0} + c^\al_\xi \right) \Gamma_\xi \dVxi \notag \\
  &= \int_{\mc{B}} \left( \delta \left(\rho^\alpha_{\xi 0} + c^\al_\xi \right) \Gamma_\xi + J_\xi \rho^\alpha_\xi \delta \Gamma_\xi \right)  \dVxi \notag \\
  &= \int_{\mc{B}} \left( \delta c^\al_\xi \Gamma_\xi + J_\xi \rho^\alpha_\xi \delta \Gamma_\xi \right)  \dVxi \notag \\
  &= \int_{\mc{B}} \left( \frac{\delta c^\al_\xi}{J_\xi} \Gamma_\xi + \rho^\alpha_\xi \delta \Gamma_\xi \right)  \dv
  \label{eq:bd128_mod2}
\end{align}
%
summing both sides of \eqref{eq:bd128_mod2} over $\alpha$ gives
\begin{align}
  \delta \int_{\mc{B}} \rho_\xi \psi_\xi \, \dv
  &= \int_{\mc{B}} \left(\rho_\xi \delta \Gamma_\xi + \sum_\alpha \frac{\delta c^\al_\xi}{J_\xi} \Gamma_\xi  \right)  \dv
  \label{eq:bd128_mod}
\end{align}
%
where we've used $\rho_\xi = \sum_\alpha \rho^\alpha_\xi$.

% ============================================================
\subsection{Variation of the kinetic energy}
\label{sec:var_T_app}
% ============================================================

The variation of kinetic energy of a using \eqref{eq:bd128_mod} is
\begin{equation}
  \delta T = \sum_\xi \int_{\mc{B}} \rho_\xi\,\mbf{v}_\xi \cdot \delta\mbf{v}_\xi + \frac{1}{2} \mbf{v}_\xi \cdot \mbf{v}_\xi \sum_\al \frac{\delta c^\al_\xi}{J_\xi} \dv.
\end{equation}
Since $\delta\mbf{v}_\xi = \mathrm{d}(\delta\mbf{x}_\xi)/\mathrm{d}t$ and $\delta\mbf{x}_\xi\vert_{t_1} = \delta\mbf{x}_\xi\vert_{t_2} = \mbf{0}$, integration by parts in time gives
\begin{equation}
  \int_{t_1}^{t_2} \delta T \dt
  = \int_{t_1}^{t_2} \sum_\xi \int_{\mc{B}} \left(-\rho_\xi\,\mbf{a}_\xi - \sum_\al \frac{\dot{c}^\alpha_\xi}{J_\xi} \mbf{v}_\xi \right) \cdot \delta\mbf{x}_\xi + \frac{1}{2} \mbf{v}_\xi \cdot \mbf{v}_\xi \sum_\al \frac{\delta c_\xi^\alpha}{J_\xi}  \dv \dt
  \label{eq:deltaT_app}
\end{equation}

\subsection{Variation of the solid internal energy}
\label{sec:var_Us_F}
% ============================================================

The solid contribution is most naturally written over the phase reference measures ${\rm d}V_{\xi_s}$. Accordingly,
%
\begin{align}
  \delta U_{\xi_s} = \sum_{\xi_s} \int_{\mc{B}_{\xi_{s0}}} \rho_{\xi_{s0}} \left[\left.\p{\psi_{\xi_s}}{\mbf{F}_{\xi_{s0}}}\right\vert_{\bar\rho_{\xi_s}} \delta \mbf{F}_{\xi_{s}} 
   + \left.\p{\psi_{\xi_s}}{\bar\rho_{\xi_s}}\right\vert_{\mbf{F}_{\xi_s}} \delta \bar\rho_{\xi_s} 
  \right] {\rm d}V_{\xi_s}.
  \label{eq:deltaU_solid}
\end{align}
%
The pushed-forward form of \eqref{eq:deltaU_solid} is
%
\begin{align}
  \delta U_{\xi_s} = \sum_{\xi_s}\int_{\mc{B}} 
    \rho_{\xi_s}\left.\p{\psi_{\xi_s}}{\bar\rho_{\xi_s}}\right\vert_{\mbf{F}_{\xi_s}} \delta \bar\rho_{\xi_s}
    - \sum_\al \frac{c^\al_{\xi_s}}{J_{\xi_s}}\left.\p{\psi_s}{\bar\rho_{\xi_s}}\right\vert_{\mbf{F}_{\xi_s}} \delta\bar\rho_{\xi_s}
    - \nabla_{\mbf x} \cdot \bs{\sigma}^\prime_{\xi_s} \cdot \delta \mbf{x}_{\xi_{s}}
   \dv.
  \label{eq:deltaU_solid_push}
\end{align}
where we've integrated by parts gives to get the standard stress divergence, i.e.\
%
\begin{equation}
  \int_{\mc{B}_{\xi_{s0}}} \rho_{\xi_{s 0}}\,\left.\p{\psi_{\xi_s}}{\mbf{F}_{\xi_s}}\right\vert_{\bar\rho_{\xi_s}} \delta \mbf{F}_{\xi_s} {\rm d}V_{\xi_s} = -\int_{\mc{B}} \nabla_{\mbf{x}} \cdot \bs\sigma'_{\xi_s} \cdot \delta\mbf{x}_{\xi_s} \dv + \text{boundary terms},
  \label{eq:stress_variation}
\end{equation}
%
where the Terzaghi effective Cauchy stress\footnote{i.e.\ density held fixed} is defined as
%
\begin{align}
  \bs\sigma'_{\xi_s} \equiv  \frac{1}{J_{\xi_s}} \rho_{\xi_{s0}}\left.\p{\psi_s}{\mbf{F}_{\xi_s}}\right\vert_{\bar\rho_{\xi_s}}\, \mbf{F}_{\xi_s}^T
  \label{eq:var_cauchy_def}
\end{align}

\subsection{Variation of the interfacial energies}
\label{sec:var_gamma}
% ============================================================

With three mobile phases (aqueous, oleic, and gas), let
\[
  \phi = \phi_a + \phi_o + \phi_g,
  \qquad
  S_a = \frac{\phi_a}{\phi},
  \qquad
  S_g = \frac{\phi_g}{\phi},
\]
so that $S_o = 1-S_a-S_g = \phi_o/\phi$.  We take the interfacial energy to be generated by a three-phase capillary potential,
\[
  U_\gamma = \int_{\mc{B}} \gamma(S_a,S_g)\,\dv,
\]
which introduces two independent capillary pressures,
\begin{equation}
  p_{c,oa} \equiv -\frac{1}{\phi}\frac{\partial\gamma}{\partial S_a} > 0,
  \qquad
  p_{c,og} \equiv -\frac{1}{\phi}\frac{\partial\gamma}{\partial S_g} > 0.
  \label{eq:pc_def}
\end{equation}
The existence of the scalar potential $\gamma(S_a,S_g)$ imposes the corresponding integrability condition on admissible capillary pressure closures,
\[
  \frac{\partial}{\partial S_g}\left(\phi p_{c,oa}\right)
  =
  \frac{\partial}{\partial S_a}\left(\phi p_{c,og}\right),
\]
with the derivatives taken at fixed $\phi$.
Using
\[
  \delta S_a = \frac{\phi_o+\phi_g}{\phi^2}\,\delta\phi_a - \frac{\phi_a}{\phi^2}(\delta\phi_o+\delta\phi_g),
  \qquad
  \delta S_g = \frac{\phi_a+\phi_o}{\phi^2}\,\delta\phi_g - \frac{\phi_g}{\phi^2}(\delta\phi_a+\delta\phi_o),
\]
together with the inter-component variation identity for the spatial fields, the variation of the interfacial energy contributes
\begin{align}
  \delta U_\gamma
  &= \int_{\mc{B}} \Bigg[
      \left(-p_{c,oa}\frac{\phi_o+\phi_g}{\phi} + p_{c,og}\frac{\phi_g}{\phi}\right)\delta\phi_a \notag \\
  &\qquad\qquad
      + \left(p_{c,oa}\frac{\phi_a}{\phi} + p_{c,og}\frac{\phi_g}{\phi}\right)\delta\phi_o \notag \\
  &\qquad\qquad
      + \left(p_{c,oa}\frac{\phi_a}{\phi} - p_{c,og}\frac{\phi_a+\phi_o}{\phi}\right)\delta\phi_g
    \Bigg] \,\dv \notag \\
  &\quad + \int_{\mc{B}} \Bigg[
      \left(-p_{c,oa}\frac{\phi_o+\phi_g}{\phi} + p_{c,og}\frac{\phi_g}{\phi}\right)\nabla_{\mbf{x}}\phi_a \cdot \delta\mbf{x}_a \notag \\
  &\qquad\qquad
      + \left(p_{c,oa}\frac{\phi_a}{\phi} + p_{c,og}\frac{\phi_g}{\phi}\right)\nabla_{\mbf{x}}\phi_o \cdot \delta\mbf{x}_o \notag \\
  &\qquad\qquad
      + \left(p_{c,oa}\frac{\phi_a}{\phi} - p_{c,og}\frac{\phi_a+\phi_o}{\phi}\right)\nabla_{\mbf{x}}\phi_g \cdot \delta\mbf{x}_g
    \Bigg] \,\dv.
  \label{eq:deltaU_fg}
\end{align}

\subsection{Variation of fluid, oleic, and gas energy}
\label{sec:var_Ufog}
% ============================================================

For each mobile phase $\xi_f \in \{a,o,g\}$, let
%
\begin{equation}
  U_{\xi_f} = \int_{\mc{B}} \rho_{\xi_f}\psi_{\xi_f} \dv,
  \label{eq:U_xi_current}
\end{equation}
%
with constitutive functions
%
\[
  \psi_{\xi_f} = \psi_{\xi_f}(\bar\rho_{\xi_f}, \eta^\alpha_{\xi}).
\]
%
%
take the variation the mobile free energy
%
\begin{align}
  \delta\psi_{\xi_f}
  &= \left.\p{\psi_{\xi_f}}{\bar\rho_{\xi_f}}\right\vert_{\eta^\al_{\xi}} \delta \bar\rho_\xi + \sum_\al
  \left.\p{\psi_\xi}{\eta^\al_\xi}\right\vert_{\bar\rho_{\xi_f},\eta^{\beta\neq\al}_\xi}
  \delta\eta^\al_\xi
  \label{eq:delta_psi_mobile}
\end{align}
%
Using \eqref{eq:bd128_mod} and collecting coefficients of the independent variations
therefore gives
%
\begin{align}
  \delta U_{\xi_f}
  &= \sum_{\xi_f \in\{a,o,g\}} \int_{\mc{B}} \left[
    \rho_{\xi_f} \left.\p{\psi_{\xi_f}}{\bar\rho_{\xi_f}}\right\vert_{\eta^\alpha_\xi}
    \delta\bar\rho_{\xi_f} + \sum_\alpha  \left(
  \left.\rho_\xi\p{\psi_{\xi_f}}{\eta^\alpha_\xi}\right\vert_{\bar\rho_{\xi_f},\eta^{\beta\neq\alpha}_\xi} \delta \eta^\al_\xi + \psi_{\xi_f} \frac{\delta c^\alpha_{\xi_f}}{J_{\xi_f}} \right)
  \right] \dv.
  \label{eq:deltaU_mobile_total}
\end{align}
Thus the mobile free energies contribute to the \(\delta\bar\rho_\xi\), \(\delta\eta^\alpha_\xi\), and \(\delta c^\alpha_\xi\) coefficients.

\subsection{Variation of constraint $C_1$: volume-fraction}
\label{sec:var_C1_app}
% ============================================================

The constraint $C_1 = \int_{\mc{B}} \lambda(\sum_\xi\phi_\xi - 1)\,\dv$ is written in the current configuration, so the natural variation is Eulerian, i.e. taken at fixed spatial point $\mbf{x}$.  Since each $\phi_\xi$ is a constituent-attached spatial field, we use the Bedford--Drumheller identity
\begin{equation}
  \left.\delta\phi_\xi\right\vert_{\mbf{x}}
  = \delta\phi_\xi - \nabla_{\mbf{x}}\phi_\xi\cdot\delta\mbf{x}_\xi,
  \label{eq:delta_phi_BD_local}
\end{equation}
to express the variation in terms of the spatial variation $\delta\phi_\xi$ and the motion variation $\delta\mbf{x}_\xi$. Therefore
\begin{align}
  \delta C_1
  &= \int_{\mc{B}} \delta\lambda \left(\sum_\xi\phi_\xi - 1\right) \dv
  + \sum_\xi \int_{\mc{B}} \lambda\left(\delta\phi_\xi - \nabla_{\mbf{x}}\phi_\xi\cdot\delta\mbf{x}_\xi\right) \dv.
  \label{eq:deltaC1_app}
\end{align}
The first term enforces the saturation constraint $\sum_\xi\phi_\xi = 1$. The remaining terms contribute $\lambda$ to each $\delta\phi_\xi$ equation and $-\lambda\nabla_{\mbf{x}}\phi_\xi$ to each momentum variation $\delta\mbf{x}_\xi$.

\subsection{Variation of constraint $C_2$: component material mass balance}
\label{sec:var_C2_app}
% ============================================================

Let
\begin{equation}
  h_\xi^\alpha \equiv J_\xi - \frac{\phi_{\xi0}\bar\rho_{\xi0} \eta^\al_{\xi 0} + c_\xi^\alpha}{\phi_\xi\bar\rho_\xi \eta_\xi^\alpha},
\end{equation}
so that
\begin{equation}
  C_2 = \sum_\xi\sum_\alpha\int_{\mc{B}_{\xi0}}\mu_\xi^\alpha h_\xi^\alpha\,dV_\xi.
  \label{eq:C2_component_measure}
\end{equation}
%
The variation of \(h_\xi^\alpha\) is
%
\begin{equation}
  \delta h_\xi^\alpha
  = \delta J_\xi
  + \frac{\phi_{\xi0}\bar\rho_{\xi0}\eta^\alpha_{\xi0} + c_\xi^\alpha}{\phi_\xi^2\bar\rho_\xi \eta^\al_\xi}\,\delta\phi_\xi
  + \frac{\phi_{\xi0}\bar\rho_{\xi0}\eta^\alpha_{\xi0} + c_\xi^\alpha}{\phi_\xi\bar\rho_\xi^2 \eta^\al_\xi}\,\delta\bar\rho_\xi
  + \frac{\phi_{\xi0}\bar\rho_{\xi0}\eta^\alpha_{\xi0} + c_\xi^\alpha}{\phi_\xi\bar\rho_\xi (\eta^\al_\xi)^2}\,\delta\eta^\al_\xi
  - \frac{1}{\phi_\xi\bar\rho_\xi \eta_\xi^\alpha}\,\delta c_\xi^\alpha;
  \label{eq:dh}
\end{equation}
%
Hence
%
\begin{align}
  \delta C_2
  &= \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
      \delta\mu_\xi^\alpha\, h_\xi^\alpha \, dV_\xi
  \notag \\
  &\quad
  + \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
      \mu_\xi^\alpha \delta J_\xi \, {\rm d}V_\xi
  \notag \\
  &\quad
  + \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
      \mu_\xi^\alpha\,\frac{\phi_{\xi0}\bar\rho_{\xi0}\eta_{\xi0}^\alpha + c_\xi^\alpha}{\phi_\xi^2\bar\rho_\xi \eta_\xi^\alpha}\,\delta\phi_\xi \, {\rm d}V_\xi
  \notag \\
  &\quad
  + \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
      \mu_\xi^\alpha\,\frac{\phi_{\xi0}\bar\rho_{\xi0}\eta_{\xi0}^\alpha + c_\xi^\alpha}{\phi_\xi \bar\rho_\xi^2 \eta^\al_\xi}\,\delta\bar\rho_\xi \, {\rm d}V_\xi
  \notag \\
  &\quad
  + \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
      \mu_\xi^\alpha\,\frac{\phi_{\xi0}\bar\rho_{\xi0}\eta_{\xi0}^\alpha + c_\xi^\alpha}{\phi_\xi \bar\rho_\xi (\eta^\al_\xi)^2}\,\delta\eta_\xi^\alpha \, {\rm d}V_\xi
  &\quad
  - \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
      \frac{\mu_\xi^\alpha}{\phi_\xi\bar\rho_\xi \eta_\xi^\alpha}\,\delta c_\xi^\alpha \, {\rm d}V_\xi.
  \label{eq:deltaC2_lines}
\end{align}
Using $\dv = J_\xi {\rm d}V_\xi$, the individual contributions push forward as follows.

\paragraph{$\delta J_\xi$ terms (from $\delta\mbf{x}_\xi$):}
\begin{align}
  \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
    \mu_\xi^\alpha \delta J_\xi \, dV_\xi
  &= \sum_\xi\sum_\alpha \int_{\mc{B}}
    \mu_\xi^\alpha\,\nabla_{\mbf{x}}\cdot\delta\mbf{x}_\xi \, \dv
  \notag \\
  &= -\sum_\xi\sum_\alpha \int_{\mc{B}}
    \nabla_{\mbf{x}}\mu_\xi^\alpha \cdot \delta\mbf{x}_\xi \, \dv
    + \text{b.t.}
  \label{eq:C2_dJ}
\end{align}

\paragraph{$\delta\phi_\xi$ terms:}
\begin{align}
  \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
    \mu_\xi^\alpha\,\frac{\phi_{\xi0}\bar\rho_{\xi0} \eta_{\xi 0}^\alpha + c_\xi^\alpha}{\phi_\xi^2\bar\rho_\xi \eta_\xi^\alpha}\,\delta\phi_\xi \, {\rm d}V_\xi
  &= \sum_\xi\sum_\alpha \int_{\mc{B}}
    \frac{\mu_\xi^\alpha}{\phi_\xi} \delta\phi_\xi \, \dv.
  \label{eq:C2_dphi_push}
\end{align}

\paragraph{$\delta\bar\rho_\xi$ terms:}
\begin{align}
  \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
    \mu_\xi^\alpha\,\frac{\phi_{\xi0}\bar\rho_{\xi0} \eta_{\xi 0}^\al + c_\xi^\alpha}{\phi_\xi \bar\rho_\xi^2 \eta_\xi^\alpha}\,\delta\bar\rho_\xi \, {\rm d}V_\xi
  &= \sum_\xi\sum_\alpha \int_{\mc{B}}
    \frac{\mu_\xi^\alpha}{\bar\rho_\xi}\delta\bar\rho_\xi \dv.
  \label{eq:C2_drho_push}
\end{align}

\paragraph{$\delta\eta_\xi^\al$ terms:}
\begin{align}
  \sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
    \mu_\xi^\alpha\,\frac{\phi_{\xi0}\bar\rho_{\xi0} \eta_{\xi 0}^\alpha + c_\xi^\alpha}{\phi_\xi \bar\rho_\xi (\eta_\xi^\alpha)^2}\,\delta\eta_\xi^\alpha \, {\rm d}V_\xi
  &= \sum_\xi\sum_\alpha \int_{\mc{B}}
    \frac{\mu_\xi^\alpha}{\eta^\al_\xi}\delta\eta^\al_\xi \dv.
  \label{eq:C2_deta_push}
\end{align}

\paragraph{$\delta c_\xi^\alpha$ terms:}
\begin{align}
  -\sum_\xi\sum_\alpha \int_{\mc{B}_{\xi0}}
    \frac{\mu_\xi^\alpha}{\phi_\xi\bar\rho_\xi \eta_\xi^\alpha}\,\delta c_\xi^\alpha \, dV_\xi
  &= -\sum_\xi\sum_\alpha \int_{\mc{B}}
    \frac{\mu_\xi^\alpha}{\phi_\xi\bar\rho_\xi \eta_\xi^\alpha}\,\frac{\delta c_\xi^\alpha}{J_\xi} \, \dv.
  \label{eq:C2_dc_push}
\end{align}

\subsection{Variation of constraint $C_3$: mass partition within phase}
\label{sec:var_C3_app}
% ============================================================

\begin{align}
  \delta C_3
  &= \sum_\xi \int_{\mc{B}} \delta\pi_\xi \left(\sum_\al \eta_\xi^\al - 1\right) \dv
  + \sum_\xi \sum_\al \int_{\mc{B}} \pi_\xi \left(\delta\eta^\al_\xi - \nabla_{\mbf{x}}\eta_\xi^\al\cdot\delta\mbf{x}_\xi\right) \dv, \notag \\
  &= \sum_\xi \int_{\mc{B}} \delta\pi_\xi \left(\sum_\al \eta_\xi^\al - 1\right) \dv
  + \sum_\xi \sum_\al \int_{\mc{B}} \pi_\xi \delta\eta^\al_\xi \dv.
  \label{eq:deltaC3_app}
\end{align}
%
The first line vanishes on the constraint surface $\sum_\alpha\eta_\xi^\alpha = 1$ and recovers it as an Euler--Lagrange equation.  The second line contributes $\pi_\xi$ to each $\delta\eta_\xi^\alpha$ Euler--Lagrange equation.

% ============================================================
\subsection{Variation of constraint $C_4$: component-wise mass conservation}
\label{sec:var_C4_app}
% ============================================================

Recalling
\begin{equation}
  C_4 = \sum_\alpha \int_{\mc{B}} \tau^\alpha \sum_\xi \frac{\dot{c}^\alpha_\xi}{J_\xi}  \,\dv,
  \label{eq:c4_var}
\end{equation}
%
we'll compute the variations term-by-term for clarity, first let's define a new spatial field as the collection
\[
  \hat{c}^\alpha_\xi(\mbf{x}) \equiv \frac{\dot{c}^\alpha_\xi}{J_\xi}.
\]
%
The variation of $\hat{c}^\alpha_\xi$ then proceeds with the identity of a variation at a fixed spatial point
%
\[
  \left.\delta\hat{c}^\alpha_\xi\right\vert_{\mbf{x}} = \delta\hat{c}^\alpha_\xi - \nabla_{\mbf{x}} \hat{c}^\alpha_\xi \cdot \delta\mbf{x}_\xi
\]
%
Now expanding the collection and computing the local variations
\begin{align}
  \delta\hat{c}^\alpha_\xi &=  \frac{\delta \dot{c}^\alpha_\xi}{J_\xi}  - \frac{\dot{c}^\alpha_\xi }{J_\xi} \nabla_\mbf{x} \cdot \delta \mbf{x}_\xi \notag \\
   &=  \frac{\delta \dot{c}^\alpha_\xi}{J_\xi}  - \hat{c}^\alpha_\xi \nabla_\mbf{x} \cdot \delta \mbf{x}_\xi \notag \\
\end{align}
%
Computing the variation of the first term in \eqref{eq:c4_var} gives
\begin{align}
  \int_{t_1}^{t_2} \int_{\mc{B}} \sum_\xi \sum_\alpha \tau^\al \left.\delta\hat{c}^\alpha_\xi \right\vert_{\mbf x} \dv \dt 
  &=  \int_{t_1}^{t_2} \int_{\mc{B}} \sum_\xi \sum_\alpha  \frac{\tau^\al}{J_\xi} \delta \dot{c}^\alpha_\xi - \tau^\al \hat{c}^\alpha_\xi \nabla_\mbf{x} \cdot \delta \mbf{x} - \tau^\al \nabla_{\mbf{x}} \hat{c}^\alpha_\xi \cdot \delta\mbf{x}_\xi \dv \dt \notag \\
   &= \int_{t_1}^{t_2} \int_{\mc{B}} \sum_\xi \sum_\alpha \frac{\tau^\al }{J_\xi} \delta \dot{c}^\alpha_\xi  - \tau^\al \nabla_\mbf{x} \cdot \left(\hat{c}^\alpha_\xi \delta \mbf{x}_\xi \right) \dv \dt \notag \\
   &= \int_{t_1}^{t_2} \int_{\mc{B}} \sum_\xi \sum_\alpha  - \frac{D_\xi \tau^\al}{Dt}\frac{\delta c^\alpha_\xi}{J_\xi}  + \frac{\dot{c}^\alpha_\xi}{J_\xi} \nabla_{\mbf x} \tau^\al \cdot  \delta \mbf{x}_\xi  \dv \dt + \text{b.t.}
   \label{eq:c4_term1}
\end{align}
%
where we've integrated by parts in time (with $\delta c_\xi^\alpha|_{t_1} = \delta c_\xi^\alpha|_{t_2} = 0$) and integrated by parts in space (supressing the boundary terms (b.t.) for brevity).  

% ============================================================
\subsection{Collected variational principle}
\label{sec:collected_var}
% ============================================================

Evaluating all variations and collecting by independent variations $\delta\mbf{x}_\xi$, $\delta\phi_\xi$, $\delta\bar\rho_\xi$, $\delta\eta^\alpha_\xi$, and $\delta c^\alpha_\xi/J_\xi$ gives
\begin{align}
  0 = & \sum_{\xi_s} \int_{\mc{B}} \Bigg[
    -\rho_{\xi_s}\mbf{a}_{\xi_s} 
    + \nabla_{\mbf{x}}\cdot\bs\sigma'_{\xi_s} + \rho_{\xi_s}\mbf{g} + \mbf{b}_{\xi_s}
    - \lambda\nabla_{\mbf{x}}\phi_{\xi_s}
    - \sum_\alpha\nabla_{\mbf{x}}\mu_{\xi_s}^\alpha
    + \sum_\alpha \hat{c}_{\xi_s}^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_{\xi_s}\right)
  \Bigg]\cdot\delta\mbf{x}_{\xi_s} \,\dv \notag \\
  &+ \int_{\mc{B}} \Bigg[
    -\rho_a\mbf{a}_a 
    + \rho_a\mbf{g} + \mbf{b}_a
    - \lambda\nabla_{\mbf{x}}\phi_a
    - \sum_\alpha\nabla_{\mbf{x}}\mu_a^\alpha
    + \sum_\alpha \hat{c}_a^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_a\right)
    + \left(p_{c,oa}\frac{\phi_o+\phi_g}{\phi} - p_{c,og}\frac{\phi_g}{\phi}\right)\nabla_{\mbf{x}}\phi_a
  \Bigg]\cdot\delta\mbf{x}_a \,\dv \notag \\
  &+ \int_{\mc{B}} \Bigg[
    -\rho_o\mbf{a}_o 
    + \rho_o\mbf{g} + \mbf{b}_o
    - \lambda\nabla_{\mbf{x}}\phi_o
    - \sum_\alpha\nabla_{\mbf{x}}\mu_o^\alpha
    + \sum_\alpha \hat{c}_o^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_o\right)
    - \left(p_{c,oa}\frac{\phi_a}{\phi} + p_{c,og}\frac{\phi_g}{\phi}\right)\nabla_{\mbf{x}}\phi_o
  \Bigg]\cdot\delta\mbf{x}_o \,\dv \notag \\
  &+ \int_{\mc{B}} \Bigg[
    -\rho_g\mbf{a}_g 
    + \rho_g\mbf{g} + \mbf{b}_g
    - \lambda\nabla_{\mbf{x}}\phi_g
    - \sum_\alpha\nabla_{\mbf{x}}\mu_g^\alpha
    + \sum_\alpha \hat{c}_g^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_g\right)
    + \left(-p_{c,oa}\frac{\phi_a}{\phi} + p_{c,og}\frac{\phi_a+\phi_o}{\phi}\right)\nabla_{\mbf{x}}\phi_g
  \Bigg]\cdot\delta\mbf{x}_g \,\dv \notag \\
&+ \sum_{\xi_s} \int_{\mc{B}} \Bigg[
  \lambda + \sum_\alpha \frac{\mu_{\xi_s}^\alpha}{\phi_{\xi_s}}
    \Bigg] \delta\phi_{\xi_s} \,\dv \notag \\
  &+ \int_{\mc{B}} \Bigg[
      \lambda + \sum_\alpha \frac{\mu_a^\alpha}{\phi_a}
      + p_{c,oa}\frac{\phi_o+\phi_g}{\phi} - p_{c,og}\frac{\phi_g}{\phi}
    \Bigg] \delta\phi_a \,\dv \notag \\
  &+ \int_{\mc{B}} \Bigg[
      \lambda + \sum_\alpha \frac{\mu_o^\alpha}{\phi_o}
      - p_{c,oa}\frac{\phi_a}{\phi} - p_{c,og}\frac{\phi_g}{\phi}
    \Bigg] \delta\phi_o \,\dv \notag \\
  &+ \int_{\mc{B}} \Bigg[
      \lambda + \sum_\alpha \frac{\mu_g^\alpha}{\phi_g}
      - p_{c,oa}\frac{\phi_a}{\phi} + p_{c,og}\frac{\phi_a+\phi_o}{\phi}
    \Bigg] \delta\phi_g \,\dv \notag \\
  &+ \sum_{\xi_s}\int_{\mc{B}} \Bigg[
    -\left(\rho_{\xi_s} - \sum_\alpha \frac{c_{\xi_s}^\alpha}{J_{\xi_s}}\right)
    \left.\p{\psi_{\xi_s}}{\bar\rho_{\xi_s}}\right\vert_{\mbf{F}_{\xi_s}}
    + \sum_\alpha\frac{\mu_{\xi_s}^\alpha}{\bar\rho_{\xi_s}}
  \Bigg] \delta\bar\rho_{\xi_s} \,\dv \notag \\
  &+ \sum_{\xi_f} \int_{\mc{B}} \Bigg[
    -\rho_{\xi_f}\left.\p{\psi_{\xi_f}}{\bar\rho_{\xi_f}}\right\vert_{\eta^\alpha_\xi}
    + \sum_\alpha\frac{\mu_{\xi_f}^\alpha}{\bar\rho_{\xi_f}}
  \Bigg] \delta\bar\rho_{\xi_f} \,\dv \notag \\
&+ \sum_{\xi_f}\sum_\alpha \int_{\mc{B}} \Bigg[
  \frac{\mu_{\xi_f}^\alpha}{\eta_\xi^\alpha} + \pi_{\xi_f}
  - \rho_{\xi_f}\left.\p{\psi_{\xi_f}}{\eta^\alpha_\xi}\right\vert_{\bar\rho_{\xi_f},\eta^{\beta\neq\alpha}_\xi}
  \Bigg] \delta\eta^\alpha_\xi \,\dv \notag \\
  &+ \sum_{\xi_s}\sum_\alpha \int_{\mc{B}} \Bigg[
    \tfrac{1}{2}\mbf{v}_{\xi_s}\cdot\mbf{v}_{\xi_s} + \mc{L}^\alpha_{\xi_s}
    - \frac{D_{\xi_s}\tau^\al}{Dt}
    - \frac{\mu_{\xi_s}^\alpha}{\phi_{\xi_s}\bar\rho_{\xi_s}\eta_{\xi_s}^\alpha}
  \Bigg] \frac{\delta c^\alpha_{\xi_s}}{J_{\xi_s}} \,\dv \notag \\
  &+ \sum_{\xi_f}\sum_\alpha \int_{\mc{B}} \Bigg[
    \tfrac{1}{2}\mbf{v}_{\xi_f}\cdot\mbf{v}_{\xi_f} + \mc{L}^\alpha_{\xi_f} + \psi_{\xi_f}
    - \frac{D_{\xi_f}\tau^\al}{Dt}
    - \frac{\mu_{\xi_f}^\alpha}{\phi_{\xi_f}\bar\rho_{\xi_f}\eta_\xi^\alpha}
  \Bigg] \frac{\delta c^\alpha_{\xi_f}}{J_{\xi_f}} \,\dv + \text{boundary terms}.
  \label{eq:collected_full}
\end{align}
The multiplier variations are omitted here because they simply return the constraint equations~\eqref{eq:C1_def}--\eqref{eq:C4_def}. Since all variations are considered arbitrary, each bracketed integrand must vanish pointwise, yielding the Euler--Lagrange equations.

\subsection{Euler--Lagrange equations}

The Euler--Lagrange equations generated from arbirtrary $\delta \phi_{\xi_s}$ gives
%
\begin{equation}
  \phi_{\xi_s} \lambda = - \sum_\alpha \mu^\al_{\xi_s}
  \label{eq:el_lambda}
\end{equation}
%
Subsitituting \eqref{eq:el_lambda} into the Euler-Lagrange equations associated with arbitrary $\delta \mbf{x}_{\xi_s}$ and rearrangeing gives the solid phases $\xi_s$ momentum balance

\begin{equation}
    \rho_{\xi_s}\mbf{a}_{\xi_s} =
    \nabla_{\mbf{x}}\cdot\bs\sigma'_{\xi_s} + \rho_{\xi_s}\mbf{g} + \mbf{b}_{\xi_s}
    + \phi_{\xi_s} \nabla_{\mbf{x}}\lambda
    + \sum_\alpha \hat{c}_{\xi_s}^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_{\xi_s}\right)
\end{equation}

Next, we rearrange the Euler-Lagrange associated with arbitrary $\delta \bar\rho_{\xi_f}$ and define
%
\begin{equation}
  \sum_\al \frac{\mu_{\xi_f}^\al}{\phi_{\xi_f}} = \bar\rho^2_{\xi_f} \p{\psi_{\xi_f}}{\bar\rho_{\xi_f}} \equiv \bar{p}_{\xi_f}
\end{equation}
%
where $\bar{p}_{\xi_f}$ is the mobile phase pressure.

The Euler--Lagrange equations associated with arbitrary $\delta\eta_{\xi_f}^\alpha$ give
\begin{equation}
  0 = \frac{\mu_{\xi_f}^\alpha}{\eta_{\xi_f}^\alpha} + \pi_{\xi_f}
  - \rho_{\xi_f}\left.\p{\psi_{\xi_f}}{\eta_{\xi_f}^\alpha}\right\vert_{\bar\rho_{\xi_f},\eta_{\xi_f}^{\beta\neq\alpha}},
  \qquad \alpha=1,\ldots,N,
  \label{eq:el_eta_mobile}
\end{equation}
while variation with respect to $\pi_{\xi_f}$ returns the composition normalization
constraint
\begin{equation}
  \sum_{\alpha=1}^N \eta_{\xi_f}^\alpha = 1.
  \label{eq:el_eta_constraint}
\end{equation}
Thus $\pi_{\xi_f}$ is the phase-level multiplier that enforces the redundant
composition coordinate.  Equation~\eqref{eq:el_eta_mobile} may be read
componentwise as
\begin{equation}
  \rho_{\xi_f}\left.\p{\psi_{\xi_f}}{\eta_{\xi_f}^\alpha}\right\vert_{\bar\rho_{\xi_f},\eta_{\xi_f}^{\beta\neq\alpha}}
  - \frac{\mu_{\xi_f}^\alpha}{\eta_{\xi_f}^\alpha}
  = \pi_{\xi_f},
  \label{eq:pi_componentwise}
\end{equation}
so the same value of $\pi_{\xi_f}$ must be obtained from every component in a
given phase.  Equivalently, one may subtract the equation for component $\beta$
from the equation for component $\alpha$ to eliminate $\pi_{\xi_f}$, leaving
$N-1$ independent composition equations together with
\eqref{eq:el_eta_constraint}.  In this sense $\pi_{\xi_f}$ is not an additional
material response; it enforces normalization and records the common value needed
for consistency among the componentwise composition equations.
%
\begin{equation}
  \rho_{\xi_f}\left.\p{\psi_{\xi_f}}{\eta_{\xi_f}^\alpha}\right\vert_{\bar\rho_{\xi_f},\eta_{\xi_f}^{\kappa\neq\alpha}}
  - \frac{\mu_{\xi_f}^\alpha}{\eta_{\xi_f}^\alpha}
  = \rho_{\xi_f}\left.\p{\psi_{\xi_f}}{\eta_{\xi_f}^\beta}\right\vert_{\bar\rho_{\xi_f},\eta_{\xi_f}^{\kappa\neq\beta}}
  - \frac{\mu_{\xi_f}^\beta}{\eta_{\xi_f}^\beta}
,
  \label{eq:pi_difference}
\end{equation}

The Euler--Lagrange equations associated with arbitrary $\delta c_\xi^\alpha$
come from the final two terms in the collected variational principle.  For the
solid phases, arbitrariness of each $\delta c_{\xi_s}^\alpha$ gives
\begin{equation}
  \frac{D_{\xi_s}\tau^\al}{Dt} = \frac{1}{2}\mbf{v}_{\xi_s}\cdot\mbf{v}_{\xi_s}
  + \mc{L}^\al_{\xi_s}
  - \frac{\mu^\alpha_{\xi_s}}{\phi_{\xi_s} \bar\rho_{\xi_s}\eta_{\xi_s}^\alpha},
  \qquad \xi_s\in\{A,B, \ldots \},
  \label{eq:el_c_solid}
\end{equation}
%
For the mobile phases,
\begin{equation}
   \frac{D_{\xi_f}\tau^\al}{Dt}
  = \tfrac{1}{2}\mbf{v}_{\xi_f}\cdot\mbf{v}_{\xi_f}
  + \mc{L}^\alpha_{\xi_f} + \psi_{\xi_f}
  - \frac{\mu_{\xi_f}^\alpha}{\phi_{\xi_f}\bar\rho_{\xi_f}\eta_{\xi_f}^\alpha},
  \qquad \xi_f\in\{a,o,g\}.
  \label{eq:el_c_mobile}
\end{equation}
These equations are the stationarity conditions conjugate to component mass
addition or removal.  The index $\alpha$ is not summed in
\eqref{eq:el_c_solid}--\eqref{eq:el_c_mobile}; rather, each component transfer
variation gives its own scalar balance.  For a single-component solid,
$\eta_{\xi_s}^\alpha=1$ in \eqref{eq:el_c_solid}.

It is useful to define the phase-specific generalized transfer potential for
component $\alpha$ by
\begin{equation}
  \mc{A}_\xi^\alpha
  \equiv \frac{D_\xi\tau^\alpha}{Dt}.
  \label{eq:generalized_transfer_potential_def}
\end{equation}
Equations~\eqref{eq:el_c_solid}--\eqref{eq:el_c_mobile} then give explicit
constitutive expressions for $\mc{A}_\xi^\alpha$ in each phase.  Interphase
equilibrium for a component is the absence of a thermodynamic driving force for
moving that component between any two present phases, i.e.\
\begin{equation}
  \mc{A}_a^\alpha = \mc{A}_o^\alpha = \mc{A}_g^\alpha
  \qquad \text{for each present component }\alpha.
  \label{eq:transfer_potential_equilibrium}
\end{equation}
In the local homogeneous, inertialess equilibrium limit, the material derivatives
$D_\xi\tau^\alpha/Dt$ are common to all present phases and may be taken to vanish
up to an arbitrary additive gauge for $\tau^\alpha$.  With the Coleman--Noll
identification of the transfer work coefficient $\mc{L}_\xi^\alpha$ and a
thermodynamically consistent phase free energy, the equality
\eqref{eq:transfer_potential_equilibrium} reduces to equality of the ordinary
component chemical potentials, equivalently equality of component fugacities.
Thus the usual flash equilibrium equations are recovered as the local-equilibrium
specialization of the $\tau^\alpha$ stationarity conditions, while
\eqref{eq:el_c_solid}--\eqref{eq:el_c_mobile} also define non-equilibrium
component-transfer driving forces.

Using the pressure definition along with the Euler-Lagrange equations associated with the arbitrary $\delta \phi_{f}$ and the definitions of saturations introduced earlier gives three equations
%
\begin{subequations}
  \begin{align}
    0 &= \lambda + \bar{p}_a + (1 - S_a) p_{c,oa} - S_g p_{c,og} \label{eq:el_bc_a} \\
    0 &= \lambda + \bar{p}_o - S_a p_{c,oa} - S_g p_{c,og} \label{eq:el_bc_b} \\
    0 &= \lambda + \bar{p}_g - S_a p_{c,oa} + (1 - S_g) p_{c,og} \label{eq:el_bc_c}
  \end{align}
  \label{eq:el_pc}
\end{subequations}
%
Subtracting \eqref{eq:el_bc_a} from \eqref{eq:el_bc_b} gives
%
\begin{equation}
  p_{c,oa} = \bar{p}_o - \bar{p}_a.
\end{equation}
%
Subtracting \eqref{eq:el_bc_c} from \eqref{eq:el_bc_b} gives
%
\begin{equation}
  p_{c,og} = \bar{p}_o - \bar{p}_g
\end{equation}
%
Summing equations~\ref{eq:el_pc}
%
\begin{align}
  0 &= 3 \lambda + \bar{p}_a + \bar{p}_o + \bar{p}_g  + (1 - 3 S_a) p_{c,oa} + (1 - 3 S_g) p_{c,og}
\end{align}
%
and solving for $\lambda$ gives
\begin{align}
  \lambda &= -\frac{1}{3} \left( \bar{p}_a + \bar{p}_o + \bar{p}_g \right)  - \left(\frac{1}{3} - S_a\right) p_{c,oa} - \left(\frac{1}{3} - S_g \right) p_{c,og} \notag \\
   &= -\frac{1}{3} \left( \bar{p}_a + \bar{p}_o + \bar{p}_g + p_{c,oa} + p_{c,og} \right)  + S_a p_{c,oa} + S_g p_{c,og} \notag \\
   &= -\bar{p}_o  + S_a (\bar{p}_o - \bar{p}_a) + S_g (\bar{p}_o - \bar{p}_g) \notag \\
   &= -S_a \bar{p}_{a} + (S_a + S_g - 1)\bar{p}_o - S_g \bar{p}_g \notag \\
   &= -\left(S_a \bar{p}_{a} + S_o \bar{p}_o + S_g \bar{p}_g\right)
\end{align}

Finally, consider the Euler--Lagrange equation associated with the arbitrary
aqueous phase displacement $\delta\mbf{x}_a$.  From the collected variational
principle, the coefficient of $\delta\mbf{x}_a$ gives
\begin{align}
  \rho_a\mbf{a}_a
  &= \rho_a\mbf{g} + \mbf{b}_a
  - \lambda\nabla_{\mbf{x}}\phi_a
  - \sum_\alpha\nabla_{\mbf{x}}\mu_a^\alpha
  + \sum_\alpha \hat{c}_a^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_a\right)
  + \left[(1-S_a)p_{c,oa} - S_g p_{c,og}\right]\nabla_{\mbf{x}}\phi_a .
  \label{eq:el_mom_a_start}
\end{align}
Using $\sum_\alpha\mu_a^\alpha = \phi_a\bar p_a$, obtained by rearranging the
definition of the mobile phase pressure, this becomes
\begin{align}
  \rho_a\mbf{a}_a
  &= \rho_a\mbf{g} + \mbf{b}_a
  - \nabla_{\mbf{x}}(\phi_a\bar p_a)
  + \sum_\alpha \hat{c}_a^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_a\right)
  + \left[(1-S_a)p_{c,oa} - S_g p_{c,og} - \lambda\right]\nabla_{\mbf{x}}\phi_a .
  \label{eq:el_mom_a_mu_sub}
\end{align}
Now substitute the capillary pressure relations
$p_{c,oa}=\bar p_o-\bar p_a$ and $p_{c,og}=\bar p_o-\bar p_g$, together with
$\lambda=-\left(S_a\bar p_a+S_o\bar p_o+S_g\bar p_g\right)$.  Since
\begin{align}
  (1-S_a)p_{c,oa} - S_g p_{c,og} - \lambda
  &= (S_o+S_g)(\bar p_o-\bar p_a)
     - S_g(\bar p_o-\bar p_g)
     + S_a\bar p_a + S_o\bar p_o + S_g\bar p_g \notag \\
  &= \bar p_a,
\end{align}
the aqueous phase momentum balance reduces to
\begin{align}
  \rho_a\mbf{a}_a
  &= \rho_a\mbf{g} + \mbf{b}_a
  - \nabla_{\mbf{x}}(\phi_a\bar p_a)
  + \bar p_a\nabla_{\mbf{x}}\phi_a
  + \sum_\alpha \hat{c}_a^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_a\right) \notag \\
  &= \rho_a\mbf{g} + \mbf{b}_a
  - \phi_a\nabla_{\mbf{x}}\bar p_a
  + \sum_\alpha \hat{c}_a^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_a\right).
  \label{eq:el_mom_a_simplified}
\end{align}
Similarly, for the oleic and gas phases, the capillary terms and the
$\lambda\nabla_{\mbf{x}}\phi_\xi$ term combine with
$-\nabla_{\mbf{x}}(\phi_\xi\bar p_\xi)$ to give the final mobile phase momentum
balances
\begin{align}
  \rho_o\mbf{a}_o
  &= \rho_o\mbf{g} + \mbf{b}_o
  - \phi_o\nabla_{\mbf{x}}\bar p_o
  + \sum_\alpha \hat{c}_o^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_o\right),
  \label{eq:el_mom_o_simplified} \\
  \rho_g\mbf{a}_g
  &= \rho_g\mbf{g} + \mbf{b}_g
  - \phi_g\nabla_{\mbf{x}}\bar p_g
  + \sum_\alpha \hat{c}_g^{\alpha}\left(\nabla_{\mbf{x}}\tau^\al - \mbf{v}_g\right).
  \label{eq:el_mom_g_simplified}
\end{align}
